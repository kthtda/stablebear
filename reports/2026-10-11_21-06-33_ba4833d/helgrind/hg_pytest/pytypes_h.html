234    Holds a reference to a Python object (no reference counting)
235
236    The `handle` class is a thin wrapper around an arbitrary Python object (i.e. a
237    ``PyObject *`` in Python's C API). It does not perform any automatic reference
238    counting and merely provides a basic C++ interface to various Python API functions.
239
240    .. seealso::
241        The `object` class inherits from `handle` and adds automatic reference
242        counting features.
243\endrst */

250    /// Not using ``handle(PyObject *ptr)`` to avoid implicit conversion from ``0``.
251    template <typename T,

261                                           std::is_convertible<T, PyObject *>>::value,
262                            int> = 0>

264    handle(T &obj) : m_ptr(obj) {}

371/** \rst
372    Holds a reference to a Python object (with reference counting)
373
374    Like `handle`, the `object` class is a thin wrapper around an arbitrary Python
375    object (i.e. a ``PyObject *`` in Python's C API). In contrast to `handle`, it
376    optionally increases the object's reference count upon construction, and it
377    *always* decreases the reference count when the `object` instance goes out of
378    scope and is destructed. When using `object` instances consistently, it is much
379    easier to get reference counting right at the first attempt.
380\endrst */
381class object : public handle {
382public:
383    object() = default;
384    PYBIND11_DEPRECATED("Use reinterpret_borrow<object>() or reinterpret_steal<object>()")
385    object(handle h, bool is_borrowed) : handle(h) {
386        if (is_borrowed) {
387            inc_ref();
388        }
389    }
390    /// Copy constructor; always increases the reference count
391    object(const object &o) : handle(o) { inc_ref(); }
392    /// Move constructor; steals the object from ``other`` and preserves its reference count
393    object(object &&other) noexcept : handle(other) { other.m_ptr = nullptr; }
394    /// Destructor; automatically calls `handle::dec_ref()`
395    ~object() { dec_ref(); }
396
397    /** \rst
398        Resets the internal pointer to ``nullptr`` without decreasing the
399        object's reference count. The function returns a raw handle to the original
400        Python object.
401    \endrst */
402    handle release() {
403        PyObject *tmp = m_ptr;
404        m_ptr = nullptr;
405        return handle(tmp);
406    }
407
408    object &operator=(const object &other) {
409        // Skip inc_ref and dec_ref if both objects are the same
410        if (!this->is(other)) {
411            other.inc_ref();
412            // Use temporary variable to ensure `*this` remains valid while
413            // `Py_XDECREF` executes, in case `*this` is accessible from Python.
414            handle temp(m_ptr);
415            m_ptr = other.m_ptr;
416            temp.dec_ref();
417        }
418        return *this;
419    }
420
421    object &operator=(object &&other) noexcept {
422        if (this != &other) {
423            handle temp(m_ptr);
424            m_ptr = other.m_ptr;
425            other.m_ptr = nullptr;
426            temp.dec_ref();
427        }
428        return *this;
429    }

513// The code in this struct is very unusual, to minimize the chances of
514// masking bugs (elsewhere) by errors during the error handling (here).
515// This is meant to be a lifeline for troubleshooting long-running processes
516// that crash under conditions that are virtually impossible to reproduce.
517// Low-level implementation alternatives are preferred to higher-level ones
518// that might raise cascading exceptions. Last-ditch-kind-of attempts are made
519// to report as much of the original error as possible, even if there are
520// secondary issues obtaining some of the details.
521struct error_fetch_and_normalize {
522    // This comment only applies to Python <= 3.11:
523    //     Immediate normalization is long-established behavior (starting with
524    //     https://github.com/pybind/pybind11/commit/135ba8deafb8bf64a15b24d1513899eb600e2011
525    //     from Sep 2016) and safest. Normalization could be deferred, but this could mask
526    //     errors elsewhere, the performance gain is very minor in typical situations
527    //     (usually the dominant bottleneck is EH unwinding), and the implementation here
528    //     would be more complex.
529    // Starting with Python 3.12, PyErr_Fetch() normalizes exceptions immediately.
530    // Any errors during normalization are tracked under __notes__.
531    explicit error_fetch_and_normalize(const char *called) {
532        PyErr_Fetch(&m_type.ptr(), &m_value.ptr(), &m_trace.ptr());
533        if (!m_type) {
534            pybind11_fail("Internal error: " + std::string(called)
535                          + " called while "
536                            "Python error indicator not set.");
537        }
538        const char *exc_type_name_orig = detail::obj_class_name(m_type.ptr());
539        if (exc_type_name_orig == nullptr) {
540            pybind11_fail("Internal error: " + std::string(called)
541                          + " failed to obtain the name "
542                            "of the original active exception type.");
543        }
544        m_lazy_error_string = exc_type_name_orig;
545#if PY_VERSION_HEX >= 0x030C0000
546        // The presence of __notes__ is likely due to exception normalization
547        // errors, although that is not necessarily true, therefore insert a
548        // hint only:
549        const int has_notes = PyObject_HasAttrString(m_value.ptr(), "__notes__");
550        if (has_notes == 1) {
551            m_lazy_error_string += "[WITH __notes__]";
552        } else if (has_notes == -1) {
553            // Ignore secondary errors when probing for __notes__ to avoid leaking a
554            // spurious exception while still reporting the original error.
555            PyErr_Clear();
556        }
557#else
558        // PyErr_NormalizeException() may change the exception type if there are cascading
559        // failures. This can potentially be extremely confusing.
560        PyErr_NormalizeException(&m_type.ptr(), &m_value.ptr(), &m_trace.ptr());
561        if (m_type.ptr() == nullptr) {
562            pybind11_fail("Internal error: " + std::string(called)
563                          + " failed to normalize the "
564                            "active exception.");
565        }
566        const char *exc_type_name_norm = detail::obj_class_name(m_type.ptr());
567        if (exc_type_name_norm == nullptr) {
568            pybind11_fail("Internal error: " + std::string(called)
569                          + " failed to obtain the name "
570                            "of the normalized active exception type.");
571        }
572        if (exc_type_name_norm != m_lazy_error_string) {
573            std::string msg = std::string(called)
574                              + ": MISMATCH of original and normalized "
575                                "active exception types: ";
576            msg += "ORIGINAL ";
577            msg += m_lazy_error_string;
578            msg += " REPLACED BY ";
579            msg += exc_type_name_norm;
580            msg += ": " + format_value_and_trace();
581            pybind11_fail(msg);
582        }
583#endif
584    }
585
586    error_fetch_and_normalize(const error_fetch_and_normalize &) = delete;
587    error_fetch_and_normalize(error_fetch_and_normalize &&) = delete;
588
589    std::string format_value_and_trace() const {
590        std::string result;
591        std::string message_error_string;
592        if (m_value) {
593            auto value_str = reinterpret_steal<object>(PyObject_Str(m_value.ptr()));
594            constexpr const char *message_unavailable_exc
595                = "<MESSAGE UNAVAILABLE DUE TO ANOTHER EXCEPTION>";
596            if (!value_str) {
597                message_error_string = detail::error_string();
598                result = message_unavailable_exc;
599            } else {
600                // Not using `value_str.cast<std::string>()`, to not potentially throw a secondary
601                // error_already_set that will then result in process termination (#4288).
602                auto value_bytes = reinterpret_steal<object>(
603                    PyUnicode_AsEncodedString(value_str.ptr(), "utf-8", "backslashreplace"));
604                if (!value_bytes) {
605                    message_error_string = detail::error_string();
606                    result = message_unavailable_exc;
607                } else {
608                    char *buffer = nullptr;
609                    Py_ssize_t length = 0;
610                    if (PyBytes_AsStringAndSize(value_bytes.ptr(), &buffer, &length) == -1) {
611                        message_error_string = detail::error_string();
612                        result = message_unavailable_exc;
613                    } else {
614                        result = std::string(buffer, static_cast<std::size_t>(length));
615                    }
616                }
617            }
618#if PY_VERSION_HEX >= 0x030B0000
619            auto notes
620                = reinterpret_steal<object>(PyObject_GetAttrString(m_value.ptr(), "__notes__"));
621            if (!notes) {
622                PyErr_Clear(); // No notes is good news.
623            } else {
624                auto len_notes = PyList_Size(notes.ptr());
625                if (len_notes < 0) {
626                    result += "\nFAILURE obtaining len(__notes__): " + detail::error_string();
627                } else {
628                    result += "\n__notes__ (len=" + std::to_string(len_notes) + "):";
629                    for (ssize_t i = 0; i < len_notes; i++) {
630                        PyObject *note = PyList_GET_ITEM(notes.ptr(), i);
631                        auto note_bytes = reinterpret_steal<object>(
632                            PyUnicode_AsEncodedString(note, "utf-8", "backslashreplace"));
633                        if (!note_bytes) {
634                            result += "\nFAILURE obtaining __notes__[" + std::to_string(i)
635                                      + "]: " + detail::error_string();
636                        } else {
637                            char *buffer = nullptr;
638                            Py_ssize_t length = 0;
639                            if (PyBytes_AsStringAndSize(note_bytes.ptr(), &buffer, &length)
640                                == -1) {
641                                result += "\nFAILURE formatting __notes__[" + std::to_string(i)
642                                          + "]: " + detail::error_string();
643                            } else {
644                                result += '\n';
645                                result += std::string(buffer, static_cast<std::size_t>(length));
646                            }
647                        }
648                    }
649                }
650            }
651#endif
652        } else {
653            result = "<MESSAGE UNAVAILABLE>";
654        }
655        if (result.empty()) {
656            result = "<EMPTY MESSAGE>";
657        }
658
659        bool have_trace = false;
660        if (m_trace) {
661#if !defined(PYPY_VERSION) && !defined(GRAALVM_PYTHON)
662            auto *tb = reinterpret_cast<PyTracebackObject *>(m_trace.ptr());
663
664            // Get the deepest trace possible.
665            while (tb->tb_next) {
666                tb = tb->tb_next;
667            }
668
669            PyFrameObject *frame = tb->tb_frame;
670            Py_XINCREF(frame);
671            result += "\n\nAt:\n";
672            while (frame) {
673                PyCodeObject *f_code = PyFrame_GetCode(frame);
674                int lineno = PyFrame_GetLineNumber(frame);
675                result += "  ";
676                result += handle(f_code->co_filename).cast<std::string>();
677                result += '(';
678                result += std::to_string(lineno);
679                result += "): ";
680                result += handle(f_code->co_name).cast<std::string>();
681                result += '\n';
682                Py_DECREF(f_code);
683                auto *b_frame = PyFrame_GetBack(frame);
684                Py_DECREF(frame);
685                frame = b_frame;
686            }
687
688            have_trace = true;
689#endif //! defined(PYPY_VERSION)
690        }
691
692        if (!message_error_string.empty()) {
693            if (!have_trace) {
694                result += '\n';
695            }
696            result += "\nMESSAGE UNAVAILABLE DUE TO EXCEPTION: " + message_error_string;
697        }
698
699        return result;
700    }
701
702    std::string const &error_string() const {
703        if (!m_lazy_error_string_completed) {
704            m_lazy_error_string += ": " + format_value_and_trace();
705            m_lazy_error_string_completed = true;
706        }
707        return m_lazy_error_string;
708    }
709
710    void restore() {
711        if (m_restore_called) {
712            pybind11_fail("Internal error: pybind11::detail::error_fetch_and_normalize::restore() "
713                          "called a second time. ORIGINAL ERROR: "
714                          + error_string());
715        }
716        PyErr_Restore(m_type.inc_ref().ptr(), m_value.inc_ref().ptr(), m_trace.inc_ref().ptr());
717        m_restore_called = true;
718    }
719
720    bool matches(handle exc) const {
721        return (PyErr_GivenExceptionMatches(m_type.ptr(), exc.ptr()) != 0);
722    }
723
724    // Not protecting these for simplicity.
725    object m_type, m_value, m_trace;
726
727private:
728    // Only protecting invariants.
729    mutable std::string m_lazy_error_string;
730    mutable bool m_lazy_error_string_completed = false;
731    mutable bool m_restore_called = false;
732};

740/// Fetch and hold an error which was already set in Python.  An instance of this is typically
741/// thrown to propagate python-side errors back through C++ which can either be caught manually or
742/// else falls back to the function dispatcher (which then raises the captured error back to
743/// python).
744class PYBIND11_EXPORT_EXCEPTION error_already_set : public std::exception {
745public:
746    /// Fetches the current Python exception (using PyErr_Fetch()), which will clear the
747    /// current Python error indicator.
748    error_already_set()
749        : m_fetched_error{new detail::error_fetch_and_normalize("pybind11::error_already_set"),
750                          m_fetched_error_deleter} {}
751
752    /// The what() result is built lazily on demand.
753    /// WARNING: This member function needs to acquire the Python GIL. This can lead to
754    ///          crashes (undefined behavior) if the Python interpreter is finalizing.
755    const char *what() const noexcept override;
756
757    /// Restores the currently-held Python error (which will clear the Python error indicator first
758    /// if already set).
759    /// NOTE: This member function will always restore the normalized exception, which may or may
760    ///       not be the original Python exception.
761    /// WARNING: The GIL must be held when this member function is called!
762    void restore() { m_fetched_error->restore(); }
763
764    /// If it is impossible to raise the currently-held error, such as in a destructor, we can
765    /// write it out using Python's unraisable hook (`sys.unraisablehook`). The error context
766    /// should be some object whose `repr()` helps identify the location of the error. Python
767    /// already knows the type and value of the error, so there is no need to repeat that.
768    void discard_as_unraisable(object err_context) {
769        restore();
770        PyErr_WriteUnraisable(err_context.ptr());
771    }
772    /// An alternate version of `discard_as_unraisable()`, where a string provides information on
773    /// the location of the error. For example, `__func__` could be helpful.
774    /// WARNING: The GIL must be held when this member function is called!
775    void discard_as_unraisable(const char *err_context) {
776        discard_as_unraisable(reinterpret_steal<object>(PYBIND11_FROM_STRING(err_context)));
777    }
778
779    // Does nothing; provided for backwards compatibility.
780    PYBIND11_DEPRECATED("Use of error_already_set.clear() is deprecated")
781    void clear() {}
782
783    /// Check if the currently trapped error type matches the given Python exception class (or a
784    /// subclass thereof).  May also be passed a tuple to search for any exception class matches in
785    /// the given tuple.
786    bool matches(handle exc) const { return m_fetched_error->matches(exc); }
787
788    const object &type() const { return m_fetched_error->m_type; }
789    const object &value() const { return m_fetched_error->m_value; }
790    const object &trace() const { return m_fetched_error->m_trace; }
791
792private:
793    std::shared_ptr<detail::error_fetch_and_normalize> m_fetched_error;
794
795    /// WARNING: This custom deleter needs to acquire the Python GIL. This can lead to
796    ///          crashes (undefined behavior) if the Python interpreter is finalizing.
797    static void m_fetched_error_deleter(detail::error_fetch_and_normalize *raw_ptr);
798};

800/// Replaces the current Python error indicator with the chosen error, performing a
801/// 'raise from' to indicate that the chosen error was caused by the original error.
802inline void raise_from(PyObject *type, const char *message) {
803    // Based on _PyErr_FormatVFromCause:
804    // https://github.com/python/cpython/blob/467ab194fc6189d9f7310c89937c51abeac56839/Python/errors.c#L405
805    // See https://github.com/pybind/pybind11/pull/2112 for details.
806    PyObject *exc = nullptr, *val = nullptr, *val2 = nullptr, *tb = nullptr;
807
808    assert(PyErr_Occurred());
809    PyErr_Fetch(&exc, &val, &tb);
810    PyErr_NormalizeException(&exc, &val, &tb);
811    if (tb != nullptr) {
812        PyException_SetTraceback(val, tb);
813        Py_DECREF(tb);
814    }
815    Py_DECREF(exc);
816    assert(!PyErr_Occurred());
817
818    PyErr_SetString(type, message);
819
820    PyErr_Fetch(&exc, &val2, &tb);
821    PyErr_NormalizeException(&exc, &val2, &tb);
822    Py_INCREF(val);
823    PyException_SetCause(val2, val);
824    PyException_SetContext(val2, val);
825    PyErr_Restore(exc, val2, tb);
826}

828/// Sets the current Python error indicator with the chosen error, performing a 'raise from'
829/// from the error contained in error_already_set to indicate that the chosen error was
830/// caused by the original error.
831inline void raise_from(error_already_set &err, PyObject *type, const char *message) {
832    err.restore();
833    raise_from(type, message);
834}

1048// Helper aliases/functions to support implicit casting of values given to python
1049// accessors/methods. When given a pyobject, this simply returns the pyobject as-is; for other C++
1050// type, the value goes through pybind11::cast(obj) to convert it to an `object`.
1051template <typename T, enable_if_t<is_pyobject<T>::value, int> = 0>
1052auto object_or_cast(T &&o) -> decltype(std::forward<T>(o)) {
1053    return std::forward<T>(o);
1054}
1055// The following casting version is implemented in cast.h:
1056template <typename T, enable_if_t<!is_pyobject<T>::value, int> = 0>
1057object object_or_cast(T &&o);

1059inline handle object_or_cast(PyObject *ptr) { return ptr; }

1064class accessor : public object_api<accessor<Policy>> {
1065    using key_type = typename Policy::key_type;
1066
1067public:
1068    accessor(handle obj, key_type key) : obj(obj), key(std::move(key)) {}
1069    accessor(const accessor &) = default;
1070    accessor(accessor &&) noexcept = default;
1071
1072    // accessor overload required to override default assignment operator (templates are not
1073    // allowed to replace default compiler-generated assignments).
1074    void operator=(const accessor &a) && { std::move(*this).operator=(handle(a)); }
1075    void operator=(const accessor &a) & { operator=(handle(a)); }

1079        Policy::set(obj, key, object_or_cast(std::forward<T>(value)));
1080    }

1082    enable_if_t<!std::is_same<accessor, remove_reference_t<T>>::value> operator=(T &&value) & {
1083        get_cache() = ensure_object(object_or_cast(std::forward<T>(value)));

1102    // NOLINTNEXTLINE(google-explicit-constructor)
1103    operator object() const { return get_cache(); }
1104    PyObject *ptr() const { return get_cache().ptr(); }
1105    template <typename T>

1107        return get_cache().template cast<T>();
1108    }

1112    static object ensure_object(handle h) { return reinterpret_borrow<object>(h); }

1114    object &get_cache() const {
1115        if (!cache) {
1116            cache = Policy::get(obj, key);
1117        }
1118        return cache;
1119    }

1122    handle obj;
1123    key_type key;
1124    mutable object cache;
1125};

1514/** \rst
1515    Wraps a Python iterator so that it can also be used as a C++ input iterator
1516
1517    Caveat: copying an iterator does not (and cannot) clone the internal
1518    state of the Python iterable. This also applies to the post-increment
1519    operator. This iterator should only be used to retrieve the current
1520    value using ``operator*()``.
1521\endrst */
1522class iterator : public object {
1523public:
1524    using iterator_category = std::input_iterator_tag;
1525    using difference_type = ssize_t;
1526    using value_type = handle;
1527    using reference = const handle; // PR #3263
1528    using pointer = const handle *;
1529
1530    PYBIND11_OBJECT_DEFAULT(iterator, object, PyIter_Check)
1531
1532    iterator &operator++() {
1533        init();
1534        advance();
1535        return *this;
1536    }
1537
1538    iterator operator++(int) {
1539        // Note: We must call init() first so that rv.value is
1540        // the same as this->value just before calling advance().
1541        // Otherwise, dereferencing the returned iterator may call
1542        // advance() again and return the 3rd item instead of the 1st.
1543        init();
1544        auto rv = *this;
1545        advance();
1546        return rv;
1547    }
1548
1549    // NOLINTNEXTLINE(readability-const-return-type) // PR #3263
1550    reference operator*() const {
1551        init();
1552        return value;
1553    }
1554
1555    pointer operator->() const {
1556        init();
1557        return &value;
1558    }
1559
1560    /** \rst
1561         The value which marks the end of the iteration. ``it == iterator::sentinel()``
1562         is equivalent to catching ``StopIteration`` in Python.
1563
1564         .. code-block:: cpp
1565
1566             void foo(py::iterator it) {
1567                 while (it != py::iterator::sentinel()) {
1568                    // use `*it`
1569                    ++it;
1570                 }
1571             }
1572    \endrst */
1573    static iterator sentinel() { return {}; }
1574
1575    friend bool operator==(const iterator &a, const iterator &b) { return a->ptr() == b->ptr(); }
1576    friend bool operator!=(const iterator &a, const iterator &b) { return a->ptr() != b->ptr(); }
1577
1578private:
1579    void init() const {
1580        if (m_ptr && !value.ptr()) {
1581            auto &self = const_cast<iterator &>(*this);
1582            self.advance();
1583        }
1584    }
1585
1586    void advance() {
1587        value = reinterpret_steal<object>(PyIter_Next(m_ptr));
1588        if (value.ptr() == nullptr && PyErr_Occurred()) {
1589            throw error_already_set();
1590        }
1591    }

1593private:
1594    object value;

1899// Converts a value to the given unsigned type.  If an error occurs, you get back (Unsigned) -1;
1900// otherwise you get back the unsigned long or unsigned long long value cast to (Unsigned).
1901// (The distinction is critically important when casting a returned -1 error value to some other
1902// unsigned type: (A)-1 != (B)-1 when A and B are unsigned types of different sizes).
1903template <typename Unsigned>
1904Unsigned as_unsigned(PyObject *o) {
1905    if (sizeof(Unsigned) <= sizeof(unsigned long)) {
1906        unsigned long v = PyLong_AsUnsignedLong(o);

1908                                                                       : (Unsigned) v;
1909    }
1910    unsigned long long v = PyLong_AsUnsignedLongLong(o);
1911    return v == static_cast<unsigned long long>(-1) && PyErr_Occurred() ? (Unsigned) -1
1912                                                                        : (Unsigned) v;
1913}
