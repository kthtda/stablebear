11    /// The shortest text that reads back as @p x, so a bound such as
12    /// 0.8999999999999999 does not show as 0.9.
13    inline std::string shortest_text(float64_t x)
14    {
15      // std::to_chars for floating point needs macOS 13.3, and wheels target
16      // macOS 11, so try increasing precision until the text reads back.
17      std::array<char, 32> buffer;
18      for (int precision = 1; precision <= 17; ++precision)
19      {
20        std::snprintf(buffer.data(), buffer.size(), "%.*g", precision, x);
21        if (std::strtod(buffer.data(), nullptr) == x)
22        {
23          break;
24        }
25      }
26
27      // %g turns to scientific notation once the exponent reaches the
28      // precision, so 906680 shows as 9.0668e+05. Like std::to_chars, use
29      // fixed notation when it is no longer.
30      std::string text = buffer.data();
31      const auto e = text.find('e');
32      if (e != std::string::npos)
33      {
34        const int exponent = std::atoi(text.c_str() + e + 1);
35        if (exponent >= 0 && (x < 0) + exponent + 1 <= static_cast<int>(text.size()))
36        {
37          std::snprintf(buffer.data(), buffer.size(), "%.0f", x);
38          text = buffer.data();
39        }
40      }
41      return text;
42    }

363  template <typename T>
364  size_t Channel<T>::snap(size_t next, float64_t t) const
365  {
366    const auto& times = held_times().values();
367    if (next == 0)
368    {
369      return 0;
370    }
371    // A query computed in floating point, such as 0.3 - 0.1, can land just
372    // below the observation it means; snap it forward. As in the tolerant
373    // comparison of APL, the distance is relative to the magnitude of the
374    // timestamps, so it scales with their rounding errors and does not
375    // depend on the unit. On large timestamps close together, it can exceed
376    // the gap, so that even a query at the previous timestamp would snap past
377    // the observation recorded there; capping it at a quarter of the gap
378    // keeps a query nearer the previous observation where it is.
379    if (next < times.size() && m_snapTolerance > 0)
380    {
381      const auto prev = times[next - 1];
382      const auto magnitude = std::max(std::abs(prev), std::abs(times[next]));
383      const auto reach = std::min(m_snapTolerance * magnitude, (times[next] - prev) / 4);
384      if (times[next] - t <= reach)
385      {
386        return next;
387      }
388    }
389    return next - 1;
390  }

512  template <typename T>
513  typename TimeSeries<T>::Domain TimeSeries<T>::domain() const
514  {
515    Domain domain{-std::numeric_limits<float64_t>::infinity(), std::numeric_limits<float64_t>::infinity(), true,
516                  true};
517    for (const auto& c : m_channels)
518    {
519      if (c.size() == 0)
520      {
521        throw std::domain_error("Cannot evaluate a channel without observations");
522      }
523      // An open end at the same time as another channel's closed one still
524      // excludes that time, and likewise for starts.
525      const auto first = convert_time(c.time(0), c.unit(), m_unit);
526      if (first > domain.first || (first == domain.first && !c.start_closed()))
527      {
528        domain.first = first;
529        domain.closedStart = c.start_closed();
530      }
531      const auto last = convert_time(c.end().value_or(c.time(c.size() - 1)), c.unit(), m_unit);
532      if (last < domain.last || (last == domain.last && !c.end_closed()))
533      {
534        domain.last = last;
535        domain.closedEnd = c.end_closed();
536      }
537    }
538    return domain;
539  }
