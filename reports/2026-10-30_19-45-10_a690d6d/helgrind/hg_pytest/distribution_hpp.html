35    {
36      long double maximum = negativeInfinity;
37      for (const auto value : row)
38        maximum = std::max(maximum, value);
39      if (!std::isfinite(maximum))

45        // lose it entirely, even when several far-tail points have equal mass.
46        value -= maximum;

313          sample_query(q, population, queryGenerator, *executor);
314        }, exec, tf::DynamicPartitioner<>(1));
315    }
316

115      if (!(value >= m_start && value < m_end))
116        return 0;
117      if (!std::isfinite(m_start) || !std::isfinite(m_end))
118        return 1;
119      const auto width = m_end - m_start;
120      if (std::isfinite(width))
121        return 1 / width;
122      // Finite endpoints can have a width exceeding the floating-point range.
123      return 0.5L / (m_end / 2 - m_start / 2);
124    }
125
126    [[nodiscard]] long double log_weight(long double value) const noexcept
127    {
128      return std::log(weight(value));
129    }
130
131  private:
132    long double m_start;
133    long double m_end;
134  };

136  class Mixture;

138  /// Any built-in distribution. Interfaces take this variant and dispatch
139  /// once per row rather than once per value.
140  using Distribution = std::variant<Gaussian, Uniform, Mixture>;
141
142  /// Weighted sum of component distributions. Coefficients are normalized to
143  /// sum to one at construction, independently of any evaluation points.
144  /// Zero-coefficient components are kept but never evaluated.
145  class Mixture
146  {
147  public:
148    explicit Mixture(const std::vector<std::pair<double, Distribution>>& terms);
149
150    [[nodiscard]] const std::vector<std::pair<double, Distribution>>& terms() const noexcept { return m_terms; }
151
152    [[nodiscard]] long double weight(long double value) const;
153
154    [[nodiscard]] long double log_weight(long double value) const;
155
156  private:
157    std::vector<std::pair<double, Distribution>> m_terms;
158  };
159
160  inline Mixture::Mixture(const std::vector<std::pair<double, Distribution>>& terms)
161  {
162    if (terms.empty())
163      throw std::invalid_argument("Mixture needs at least one component");
164    double maximum = 0;
165    for (const auto& [coefficient, component] : terms)
166    {
167      if (!std::isfinite(coefficient) || coefficient < 0)
168        throw std::invalid_argument("Mixture coefficients must be finite and nonnegative");
169      maximum = std::max(maximum, coefficient);
170    }
171    if (maximum == 0)
172      throw std::invalid_argument("Mixture needs at least one positive coefficient");
173    // Scale by the maximum first so the sum cannot overflow.
174    double total = 0;
175    for (const auto& [coefficient, component] : terms)
176      total += m_terms.emplace_back(coefficient / maximum, component).first;
177    for (auto& term : m_terms)
178      term.first /= total;
179  }
180
181  inline long double Mixture::weight(long double value) const
182  {
183    long double sum = 0;
184    for (const auto& [coefficient, component] : m_terms)
185    {
186      if (coefficient > 0)
187      {
188        sum += coefficient * std::visit([&](const auto& d) { return d.weight(value); }, component);
189      }
190    }
191    return sum;
192  }
193
194  inline long double Mixture::log_weight(long double value) const
195  {
196    long double result = detail::negativeInfinity;
197    for (const auto& [coefficient, component] : m_terms)
198    {
199      if (coefficient > 0)
200      {
201        result = detail::log_add(result, std::log(coefficient)
202          + std::visit([&](const auto& d) { return d.log_weight(value); }, component));
203      }
204    }
205    return result;
206  }
207
208  /// Fill @p row with the log probabilities of @p values under @p distribution,
209  /// normalized over the values to sum to one. Tiny positive weights stay
210  /// finite. Infinite values and values outside the support get -inf; if no
211  /// value has support, the row remains all -inf.
212  inline void log_weights(const Distribution& distribution, std::span<const long double> values,
213                          std::span<long double> row)
214  {
215    if (row.size() != values.size())
216      throw std::invalid_argument("value and weight rows must have equal length");
217    if (std::ranges::any_of(values, [](long double v) { return std::isnan(v); }))
218      throw std::invalid_argument("values must not be NaN");
219    std::visit([&](const auto& d) {
220      for (size_t i = 0; i < values.size(); ++i)
221        row[i] = std::isinf(values[i]) ? detail::negativeInfinity : d.log_weight(values[i]);
222    }, distribution);
223    detail::normalize_log_row(row);
224  }
