29  /// Map a @p target in [0, cdf.back()] to the index whose CDF interval
30  /// contains it. uniform_real_distribution may round up to exactly
31  /// cdf.back() (LWG 2524); such a target is pulled back inside the range,
32  /// so the draw lands on the last positive-weight interval — never past it
33  /// onto a trailing zero-weight (ineligible) point.
34  template <typename T>
35  size_t index_for_target(std::span<const T> cdf, T target)
36  {
37    if (target >= cdf.back())
38    {
39      target = std::nextafter(cdf.back(), T(0));
40    }
41    const auto it = std::upper_bound(cdf.begin(), cdf.end(), target);
42    return static_cast<size_t>(it - cdf.begin());
43  }

48  {
49    std::uniform_real_distribution<T> uniform(T(0), cdf.back());

53  /// Replace normalized log probabilities (finite or -inf, maximum 0) with
54  /// the prefix sums of their probabilities. An all -inf row is a valid empty
55  /// region and must not be drawn from.
56  template <typename T>
57  void log_probabilities_to_cdf(std::span<T> row)
58  {
59    T total = T(0);
60    for (T &w : row)
61    {
62      total += std::exp(w);
63      w = total;
64    }
65  }
