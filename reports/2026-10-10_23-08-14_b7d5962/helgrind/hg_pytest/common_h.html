555/// Approach used to cast a previously unknown C++ instance into a Python object
556enum class return_value_policy : uint8_t {
557    /** This is the default return value policy, which falls back to the policy
558        return_value_policy::take_ownership when the return value is a pointer.
559        Otherwise, it uses return_value::move or return_value::copy for rvalue
560        and lvalue references, respectively. See below for a description of what
561        all of these different policies do. */
562    automatic = 0,
563
564    /** As above, but use policy return_value_policy::reference when the return
565        value is a pointer. This is the default conversion policy for function
566        arguments when calling Python functions manually from C++ code (i.e. via
567        handle::operator()). You probably won't need to use this. */
568    automatic_reference,
569
570    /** Reference an existing object (i.e. do not create a new copy) and take
571        ownership. Python will call the destructor and delete operator when the
572        object's reference count reaches zero. Undefined behavior ensues when
573        the C++ side does the same.. */
574    take_ownership,
575
576    /** Create a new copy of the returned object, which will be owned by
577        Python. This policy is comparably safe because the lifetimes of the two
578        instances are decoupled. */
579    copy,
580
581    /** Use std::move to move the return value contents into a new instance
582        that will be owned by Python. This policy is comparably safe because the
583        lifetimes of the two instances (move source and destination) are
584        decoupled. */
585    move,
586
587    /** Reference an existing object, but do not take ownership. The C++ side
588        is responsible for managing the object's lifetime and deallocating it
589        when it is no longer used. Warning: undefined behavior will ensue when
590        the C++ side deletes an object that is still referenced and used by
591        Python. */
592    reference,
593
594    /** This policy only applies to methods and properties. It references the
595        object without taking ownership similar to the above
596        return_value_policy::reference policy. In contrast to that policy, the
597        function or property's implicit this argument (called the parent) is
598        considered to be the owner of the return value (the child).
599        pybind11 then couples the lifetime of the parent to the child via a
600        reference relationship that ensures that the parent cannot be garbage
601        collected while Python is still using the child. More advanced
602        variations of this scheme are also possible using combinations of
603        return_value_policy::reference and the keep_alive call policy */
604    reference_internal
605};

635/// The 'instance' type which needs to be standard layout (need to be able to use 'offsetof')
636struct instance {
637    PyObject_HEAD
638    /// Storage for pointers and holder; see simple_layout, below, for a description
639    union {
640        void *simple_value_holder[1 + instance_simple_holder_in_ptrs()];
641        nonsimple_values_and_holders nonsimple;
642    };
643    /// Weak references
644    PyObject *weakrefs;
645    /// If true, the pointer is owned which means we're free to manage it with a holder.
646    bool owned : 1;
647    /**
648     * An instance has two possible value/holder layouts.
649     *
650     * Simple layout (when this flag is true), means the `simple_value_holder` is set with a
651     * pointer and the holder object governing that pointer, i.e. [val1*][holder].  This layout is
652     * applied whenever there is no python-side multiple inheritance of bound C++ types *and* the
653     * type's holder will fit in the default space (which is large enough to hold either a
654     * std::unique_ptr or std::shared_ptr).
655     *
656     * Non-simple layout applies when using custom holders that require more space than
657     * `shared_ptr` (which is typically the size of two pointers), or when multiple inheritance is
658     * used on the python side.  Non-simple layout allocates the required amount of memory to have
659     * multiple bound C++ classes as parents.  Under this layout, `nonsimple.values_and_holders` is
660     * set to a pointer to allocated space of the required space to hold a sequence of value
661     * pointers and holders followed `status`, a set of bit flags (1 byte each), i.e.
662     * [val1*][holder1][val2*][holder2]...[bb...]  where each [block] is rounded up to a multiple
663     * of `sizeof(void *)`.  `nonsimple.status` is, for convenience, a pointer to the beginning of
664     * the [bb...] block (but not independently allocated).
665     *
666     * Status bits indicate whether the associated holder is constructed (&
667     * status_holder_constructed) and whether the value pointer is registered (&
668     * status_instance_registered) in `registered_instances`.
669     */
670    bool simple_layout : 1;
671    /// For simple layout, tracks whether the holder has been constructed
672    bool simple_holder_constructed : 1;
673    /// For simple layout, tracks whether the instance is registered in `registered_instances`
674    bool simple_instance_registered : 1;
675    /// If true, get_internals().patients has an entry for this object
676    bool has_patients : 1;
677    /// If true, this Python object needs to be kept alive for the lifetime of the C++ value.
678    bool is_alias : 1;
679
680    /// Initializes all of the above type/values/holders data (but not the instance values
681    /// themselves)
682    void allocate_layout();
683
684    /// Destroys/deallocates all of the above
685    void deallocate_layout();
686
687    /// Returns the value_and_holder wrapper for the given type (or the first, if `find_type`
688    /// omitted).  Returns a default-constructed (with `.inst = nullptr`) object on failure if
689    /// `throw_if_missing` is false.
690    value_and_holder get_value_and_holder(const type_info *find_type = nullptr,
691                                          bool throw_if_missing = true);
692
693    /// Bit values for the non-simple status flags
694    static constexpr uint8_t status_holder_constructed = 1;
695    static constexpr uint8_t status_instance_registered = 2;
696};

749// Comments about the algorithm below.
750//
751// Credit: This is based on an algorithm by taocpp here:
752//    https://github.com/taocpp/sequences/blob/main/include/tao/seq/make_integer_sequence.hpp
753// but significantly simplified.
754//
755// We build up a sequence S by repeatedly doubling its length and sometimes adding 1 to the end.
756// E.g. if the current S is 0...3, then we either go to 0...7 or 0...8 on the next pass.
757// The goal is to end with S = 0...N-1.
758// The key insight is that the times we need to add an additional digit to S correspond
759// exactly to the 1's in the binary representation of the number N.
760//
761// Invariants:
762// - digit is a power of 2
763// - N_digit_is_1 is whether N's binary representation has a 1 in that digit's position.
764// - end <= N
765// - S is 0...end-1.
766// - if digit > 0, end * digit * 2 <= N < (end+1) * digit * 2
767//
768// The process starts with digit > N, end = 0, and S is empty.
769// The process concludes with digit=0, in which case, end == N and S is 0...N-1.
770
771template <size_t digit, bool N_digit_is_1, size_t N, size_t end, size_t... S> // N_digit_is_1=false
772struct make_index_sequence_impl
773    : make_index_sequence_impl<digit / 2, (N & (digit / 2)) != 0, N, 2 * end, S..., (S + end)...> {
774};
775template <size_t digit, size_t N, size_t end, size_t... S>
776struct make_index_sequence_impl<digit, true, N, end, S...>
777    : make_index_sequence_impl<digit / 2,
778                               (N & (digit / 2)) != 0,
779                               N,
780                               2 * end + 1,
781                               S...,
782                               (S + end)...,
783                               2 * end> {};
784template <size_t N, size_t end, size_t... S>
785struct make_index_sequence_impl<0, false, N, end, S...> {
786    using type = index_sequence<S...>;
787};
788constexpr size_t next_power_of_2(size_t N) { return N == 0 ? 1 : next_power_of_2(N >> 1) << 1; }
789template <size_t N>
790using make_index_sequence =
791    typename make_index_sequence_impl<next_power_of_2(N), false, N, 0>::type;
792#endif

1305// Adaptor for converting arbitrary container arguments into a vector; implicitly convertible from
1306// any standard container (or C-style array) supporting std::begin/std::end, any singleton
1307// arithmetic type (if T is arithmetic), or explicitly constructible from an iterator pair.
1308template <typename T>
1309class any_container {
1310    std::vector<T> v;
1311
1312public:
1313    any_container() = default;
1314
1315    // Can construct from a pair of iterators
1316    template <typename It, typename = enable_if_t<is_input_iterator<It>::value>>
1317    any_container(It first, It last) : v(first, last) {}
1318
1319    // Implicit conversion constructor from any arbitrary container type
1320    // with values convertible to T
1321    template <typename Container,
1322              typename = enable_if_t<
1323                  std::is_convertible<decltype(*std::begin(std::declval<const Container &>())),
1324                                      T>::value>>
1325    // NOLINTNEXTLINE(google-explicit-constructor)
1326    any_container(const Container &c) : any_container(std::begin(c), std::end(c)) {}
1327
1328    // initializer_list's aren't deducible, so don't get matched by the above template;
1329    // we need this to explicitly allow implicit conversion from one:
1330    template <typename TIn, typename = enable_if_t<std::is_convertible<TIn, T>::value>>
1331    any_container(const std::initializer_list<TIn> &c) : any_container(c.begin(), c.end()) {}
1332
1333    // Avoid copying if given an rvalue vector of the correct type.
1334    // NOLINTNEXTLINE(google-explicit-constructor)
1335    any_container(std::vector<T> &&v) : v(std::move(v)) {}
1336
1337    // Moves the vector out of an rvalue any_container
1338    // NOLINTNEXTLINE(google-explicit-constructor)
1339    operator std::vector<T> &&() && { return std::move(v); }
1340
1341    // Dereferencing obtains a reference to the underlying vector
1342    std::vector<T> &operator*() { return v; }
1343    const std::vector<T> &operator*() const { return v; }
1344
1345    // -> lets you call methods on the underlying vector
1346    std::vector<T> *operator->() { return &v; }
1347    const std::vector<T> *operator->() const { return &v; }
1348};
