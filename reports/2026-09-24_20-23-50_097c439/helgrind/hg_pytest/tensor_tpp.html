284  template <typename U, TensorProperties OtherProperties>
285  requires std::is_constructible_v<T, U>
286  void Tensor<T, Properties>::assign_from(const Tensor<U, OtherProperties>& rhs)
287  {
288    auto target = shape();
289    auto out_shape = broadcast_shapes(target, rhs.shape());
290    if (out_shape != target)
291    {
292      throw std::invalid_argument("Cannot broadcast RHS of shape " + shape_to_string(rhs.shape()) +
293          " into target of shape " + shape_to_string(target) +
294          " (broadcast result would have shape " + shape_to_string(out_shape) + ")");
295    }
296
297    auto rhs_view = rhs.broadcast_to(target);
298
299    // If the RHS aliases this tensor's storage (e.g. a[:] = a[::-1] or
300    // a[1:] = a[:-1], where both sides are views of the same buffer), an
301    // element-wise walk would read positions it has already overwritten.
302    // Materialize an independent copy of the RHS first, matching NumPy's
303    // handling of overlapping assignment. Aliasing is only possible when the
304    // element types match and the two views share the same base buffer; data()
305    // includes the view offset, so compare the buffer base (data() - offset()).
306    if constexpr (std::is_same_v<T, U> && !IsIndexed
307      && !Tensor<U, OtherProperties>::IsIndexed)
308    {
309      if (this->data() - this->offset() == rhs.data() - rhs.offset())
310      {
311        Tensor<T> rhs_copy = rhs_view.copy();
312        sb::walk(*this, [this, &rhs_copy](const std::vector<size_t>& idx){
313          // store_copy as well: rhs_copy.copy() duplicates only the outer
314          // buffer, so element types that share a buffer (point clouds, the
315          // matrix types) would still alias the source until copied per cell.
316          (*this)(idx) = detail::materialized_store_copy(rhs_copy(idx));
317        });
318        return;
319      }
320    }
321
322    sb::walk(*this, [this, &rhs_view](const std::vector<size_t>& idx){
323      // store_copy deep-copies element types that share a buffer (point clouds
324      // and the matrix types), so a slice assignment never aliases the source.
325      if constexpr (IsIndexed)
326        writable_at(idx) = detail::materialized_store_copy(T(rhs_view(idx)));
327      else
328        (*this)(idx) = detail::materialized_store_copy(T(rhs_view(idx)));
329    });
330  }

705  typename Tensor<T, Properties>::source_tensor_type masked_select(
706      const Tensor<T, Properties>& src, const Tensor<bool>& mask)
707  {
708    using TResult = typename Tensor<T, Properties>::source_tensor_type;
709    if (mask.shape() != src.shape())
710    {
711      throw std::invalid_argument(
712        "masked_select: mask shape does not match tensor shape");
713    }
714
715    // Single pass: collect matching elements into a buffer
716    std::vector<T> buf;
717    walk(src, [&](const std::vector<size_t>& idx) {
718      if (mask(idx))
719        buf.push_back(detail::materialized_store_copy(src(idx)));
720    });
721
722    if (buf.empty())
723    {
724      return TResult({0});
725    }
726
727    TResult result({buf.size()});
728    std::move(buf.begin(), buf.end(), result.data());
729
730    return result;
731  }

970    struct ResolvedSelector
971    {
972      size_t axis;
973      std::vector<size_t> indices;  // selected positions along this axis
974    };
975
976    inline std::vector<ResolvedSelector> resolve_selectors(
977      const std::vector<std::pair<size_t, AxisSelector>>& selectors,
978      const std::vector<size_t>& shape)
979    {
980      std::vector<ResolvedSelector> result;
981      result.reserve(selectors.size());
982      for (const auto& [axis, sel] : selectors)
983      {
984        if (axis >= shape.size())
985          throw std::invalid_argument("outer indexing: axis " + std::to_string(axis) +
986            " out of range for tensor with " + std::to_string(shape.size()) + " dimensions");
987
988        ResolvedSelector rs;
989        rs.axis = axis;
990        std::visit([&](const auto& idx_tensor) {
991          using IdxT = std::decay_t<decltype(idx_tensor)>;
992          if (idx_tensor.shape().size() != 1)
993            throw std::invalid_argument("outer indexing: selector must be 1D");
994
995          if constexpr (std::is_same_v<IdxT, Tensor<bool>>)
996          {
997            if (idx_tensor.shape()[0] != shape[axis])
998              throw std::invalid_argument("outer indexing: bool mask length must match axis size");
999            rs.indices = collect_true_indices(idx_tensor);
1000          }
1001          else
1002          {
1003            const auto axis_size = static_cast<ptrdiff_t>(shape[axis]);
1004            rs.indices.reserve(idx_tensor.shape()[0]);
1005            for (size_t i = 0; i < idx_tensor.shape()[0]; ++i)
1006            {
1007              // Resolve negatives against the axis size and bounds-check so
1008              // the C++ entry point is memory-safe regardless of caller.
1009              auto v = static_cast<ptrdiff_t>(idx_tensor({i}));
1010              if (v < 0) v += axis_size;
1011              if (v < 0 || v >= axis_size)
1012                throw std::out_of_range(
1013                  "index " + std::to_string(static_cast<ptrdiff_t>(idx_tensor({i}))) +
1014                  " is out of bounds for axis with size " + std::to_string(axis_size));
1015              rs.indices.push_back(static_cast<size_t>(v));
1016            }
1017          }
1018        }, sel);
1019        result.push_back(std::move(rs));
1020      }
1021      return result;
1022    }
1023  }

1131  template <typename T, TensorProperties Properties>
1132  typename Tensor<T, Properties>::source_tensor_type concatenate(
1133      const std::vector<Tensor<T, Properties>>& tensors, size_t axis)
1134  {
1135    using TResult = typename Tensor<T, Properties>::source_tensor_type;
1136    if (tensors.empty())
1137      throw std::invalid_argument("concatenate: need at least one tensor");
1138
1139    auto ndim = tensors[0].shape().size();
1140    if (axis >= ndim)
1141      throw std::invalid_argument("concatenate: axis " + std::to_string(axis) +
1142        " out of range for " + std::to_string(ndim) + "-D tensor");
1143
1144    // Validate shapes match on all axes except the join axis
1145    for (size_t t = 1; t < tensors.size(); ++t)
1146    {
1147      if (tensors[t].shape().size() != ndim)
1148        throw std::invalid_argument("concatenate: all tensors must have the same number of dimensions");
1149      for (size_t d = 0; d < ndim; ++d)
1150      {
1151        if (d != axis && tensors[t].shape()[d] != tensors[0].shape()[d])
1152          throw std::invalid_argument("concatenate: shape mismatch on dimension " + std::to_string(d));
1153      }
1154    }
1155
1156    // Compute output shape
1157    auto out_shape = tensors[0].shape();
1158    out_shape[axis] = 0;
1159    for (const auto& t : tensors)
1160      out_shape[axis] += t.shape()[axis];
1161
1162    TResult result(out_shape);
1163
1164    // Copy each tensor's data into the result
1165    size_t offset = 0;
1166    for (const auto& src : tensors)
1167    {
1168      auto src_axis_size = src.shape()[axis];
1169      walk(src, [&](const std::vector<size_t>& src_idx) {
1170        auto dst_idx = src_idx;
1171        dst_idx[axis] += offset;
1172        result(dst_idx) = detail::materialized_store_copy(src(src_idx));
1173      });
1174      offset += src_axis_size;
1175    }
1176
1177    return result;
1178  }

1180  template <typename T, TensorProperties Properties>
1181  typename Tensor<T, Properties>::source_tensor_type stack(
1182      const std::vector<Tensor<T, Properties>>& tensors, ptrdiff_t axis)
1183  {
1184    if (tensors.empty())
1185      throw std::invalid_argument("stack: need at least one tensor");
1186
1187    auto ndim = static_cast<ptrdiff_t>(tensors[0].shape().size());
1188
1189    // Resolve negative axis; valid range for new axis is [-(ndim+1), ndim]
1190    if (axis < 0)
1191      axis += ndim + 1;
1192    if (axis < 0 || axis > ndim)
1193      throw std::invalid_argument("stack: axis out of range");
1194
1195    // Validate all shapes match
1196    for (size_t t = 1; t < tensors.size(); ++t)
1197    {
1198      if (tensors[t].shape() != tensors[0].shape())
1199        throw std::invalid_argument("stack: all tensors must have the same shape");
1200    }
1201
1202    // expand_dims each tensor, then concatenate along the new axis
1203    std::vector<Tensor<T, Properties>> expanded;
1204    expanded.reserve(tensors.size());
1205    for (const auto& t : tensors)
1206      expanded.push_back(t.expand_dims(axis));
1207
1208    return concatenate(expanded, static_cast<size_t>(axis));
1209  }

1211  template <typename T, TensorProperties Properties>
1212  std::vector<Tensor<T, Properties>> split(const Tensor<T, Properties>& tensor,
1213    const std::vector<size_t>& split_points, size_t axis)
1214  {
1215    auto ndim = tensor.shape().size();
1216    if (axis >= ndim)
1217      throw std::invalid_argument("split: axis " + std::to_string(axis) +
1218        " out of range for " + std::to_string(ndim) + "-D tensor");
1219
1220    auto axis_size = tensor.shape()[axis];
1221
1222    // Build slice boundaries: [0, sp0, sp1, ..., axis_size]
1223    std::vector<size_t> boundaries;
1224    boundaries.reserve(split_points.size() + 2);
1225    boundaries.push_back(0);
1226    for (auto sp : split_points)
1227    {
1228      if (sp > axis_size)
1229        sp = axis_size;
1230      boundaries.push_back(sp);
1231    }
1232    boundaries.push_back(axis_size);
1233
1234    std::vector<Tensor<T, Properties>> result;
1235    result.reserve(boundaries.size() - 1);
1236    for (size_t i = 0; i + 1 < boundaries.size(); ++i)
1237    {
1238      std::vector<Slice> slices(ndim, all());
1239      slices[axis] = range(
1240        static_cast<ptrdiff_t>(boundaries[i]),
1241        static_cast<ptrdiff_t>(boundaries[i + 1]),
1242        std::nullopt);
1243      result.push_back(tensor[slices]);
1244    }
1245    return result;
1246  }

1250    size_t n_sections, size_t axis)

1254      throw std::invalid_argument("split: axis " + std::to_string(axis) +
1255        " out of range for " + std::to_string(ndim) + "-D tensor");

1262      throw std::invalid_argument("split: tensor of size " +

1264        " cannot be evenly split into " + std::to_string(n_sections) + " sections");

1266    auto section_size = axis_size / n_sections;

1268    split_points.reserve(n_sections - 1);
1269    for (size_t i = 1; i < n_sections; ++i)
1270      split_points.push_back(i * section_size);
1271
1272    return split(tensor, split_points, axis);

1275  template <typename T, TensorProperties Properties>
1276  std::vector<Tensor<T, Properties>> array_split(const Tensor<T, Properties>& tensor,
1277    size_t n_sections, size_t axis)
1278  {
1279    auto ndim = tensor.shape().size();
1280    if (axis >= ndim)
1281      throw std::invalid_argument("array_split: axis " + std::to_string(axis) +
1282        " out of range for " + std::to_string(ndim) + "-D tensor");
1283    if (n_sections == 0)
1284      throw std::invalid_argument("array_split: number of sections must be > 0");
1285
1286    auto axis_size = tensor.shape()[axis];
1287    auto base_size = axis_size / n_sections;
1288    auto remainder = axis_size % n_sections;
1289
1290    std::vector<size_t> split_points;
1291    split_points.reserve(n_sections - 1);
1292    size_t pos = 0;
1293    for (size_t i = 0; i + 1 < n_sections; ++i)
1294    {

1518  template <typename T, TensorProperties Properties>
1519  Tensor<T, Properties> Tensor<T, Properties>::reshape(const std::vector<ptrdiff_t>& new_shape) const
1520  {
1521    if constexpr (IsIndexed)
1522    {
1523      return Tensor(m_indexedState, m_indexedLocator.reshape(new_shape));
1524    }
1525    else
1526    {
1527      auto total = get_total_size();
1528
1529      // Resolve -1 dimension
1530      std::vector<size_t> resolved(new_shape.size());
1531      ptrdiff_t infer_idx = -1;
1532      size_t known_product = 1;
1533      for (size_t i = 0; i < new_shape.size(); ++i)
1534      {
1535        if (new_shape[i] == -1)
1536        {
1537          if (infer_idx >= 0)
1538            throw std::invalid_argument("Only one dimension can be inferred (-1)");
1539          infer_idx = static_cast<ptrdiff_t>(i);
1540        }
1541        else if (new_shape[i] < 0)
1542        {
1543          throw std::invalid_argument("Invalid dimension size: " + std::to_string(new_shape[i]));
1544        }
1545        else
1546        {
1547          resolved[i] = static_cast<size_t>(new_shape[i]);
1548          known_product *= resolved[i];
1549        }
1550      }
1551
1552      if (infer_idx >= 0)
1553      {
1554        if (known_product == 0 || total % known_product != 0)
1555          throw std::invalid_argument("Cannot infer dimension: total size " +
1556            std::to_string(total) + " is not divisible by " + std::to_string(known_product));
1557        resolved[infer_idx] = total / known_product;
1558      }
1559
1560      // Validate total size
1561      size_t new_total = 1;
1562      for (auto d : resolved) new_total *= d;
1563      if (new_total != total)
1564        throw std::invalid_argument("Cannot reshape tensor of size " +
1565          std::to_string(total) + " into shape " + shape_to_string(resolved));
1566
1567      // Use contiguous source (copy if needed)
1568      Tensor<T, Properties> src = m_isContiguous ? *this : copy();
1569
1570      Tensor<T, Properties> ret;
1571      ret.m_data = src.m_data;
1572      ret.m_offset = src.m_offset;
1573      ret.m_shape = resolved;
1574      ret.m_isContiguous = true;
1575      ret.m_viewType = ViewType::Base;
1576
1577      // Compute row-major strides
1578      ret.m_strides.resize(resolved.size());
1579      if (!resolved.empty())
1580      {
1581        ret.m_strides.back() = 1;
1582        for (auto i = static_cast<ptrdiff_t>(resolved.size()) - 2; i >= 0; --i)
1583          ret.m_strides[i] = ret.m_strides[i + 1] * static_cast<ptrdiff_t>(resolved[i + 1]);
1584      }
1585
1586      return ret;
1587    }
1588  }

1806  template <typename T, TensorProperties Properties>
1807  template <typename SliceVector>
1808  Tensor<T, Properties> Tensor<T, Properties>::extract(SliceVector sliceVector) const
1809  {
1810    Tensor ret;
1811
1812    ret.m_data = m_data; // view
1813
1814    // temporary just to get sizes
1815    ret.m_shape = m_shape;
1816    ret.m_strides = m_strides;
1817
1818    ret.m_offset = m_offset;
1819
1820    std::vector<size_t> dimsToDrop;
1821
1822    // For now, we'll drop the assumption that the tensor is contiguous in memory
1823    // as soon as we extract a subtensor. There are, however, some cases where the resulting subtensor would be
1824    // contiguous that we can try to optimize for in the future (e.g., extracting the top n rows of a matrix).
1825    // Things will work fine with this assumption dropped but certain operations could be a little slower (probably
1826    // unlikely to matter for the type of things we're targeting).
1827    ret.m_isContiguous = false;
1828
1829    size_t i = 0;
1830    for (auto & slice : sliceVector)
1831    {
1832      std::visit([i, &ret, &dimsToDrop](auto&& arg) {
1833        using argT = std::decay_t<decltype(arg)>;
1834        if constexpr (std::is_same_v<argT, SliceIndex>)
1835        {
1836          // Resolve a (possibly negative) single integer index against the
1837          // dimension size and bounds-check it, mirroring NumPy semantics.
1838          // Throwing here (rather than computing an out-of-buffer offset)
1839          // turns silent OOB reads / SIGSEGV-on-write into a clean IndexError.
1840          auto dim_size = static_cast<ptrdiff_t>(ret.m_shape[i]);
1841          auto ix = arg.index;
1842          if (ix < 0) ix += dim_size;
1843          if (ix < 0 || ix >= dim_size)
1844          {
1845            throw std::out_of_range(
1846              "index " + std::to_string(arg.index) +
1847              " is out of bounds for axis " + std::to_string(i) +
1848              " with size " + std::to_string(dim_size));
1849          }
1850          ret.m_shape[i] = 1;
1851          ret.m_offset += ix * ret.m_strides[i];
1852          dimsToDrop.emplace_back(i);
1853        }
1854        else if constexpr (std::is_same_v<argT, SliceRange>)
1855        {
1856          if (!arg.step)
1857          {
1858            arg.step = 1_z;
1859          }
1860
1861          auto step = *arg.step;
1862          if (step == 0)
1863          {
1864            throw std::invalid_argument("slice step cannot be zero");
1865          }
1866
1867          auto dim_size = static_cast<ptrdiff_t>(ret.m_shape[i]);
1868
1869          // Mirror Python's slice.indices(n): resolve negative bounds by
1870          // adding dim_size, then clamp into the valid range. For a positive
1871          // step the range is [0, n]; for a negative step it is [-1, n-1].
1872          const ptrdiff_t lower = (step < 0) ? -1_z : 0_z;
1873          const ptrdiff_t upper = (step < 0) ? dim_size - 1_z : dim_size;
1874
1875          ptrdiff_t start;
1876          if (!arg.start)
1877          {
1878            start = (step < 0) ? upper : lower;
1879          }
1880          else
1881          {
1882            start = *arg.start;
1883            if (start < 0)
1884            {
1885              start += dim_size;
1886              if (start < lower) start = lower;
1887            }
1888            else if (start > upper)
1889            {
1890              start = upper;
1891            }
1892          }
1893
1894          ptrdiff_t stop;
1895          if (!arg.stop)
1896          {
1897            stop = (step < 0) ? lower : upper;
1898          }
1899          else
1900          {
1901            stop = *arg.stop;
1902            if (stop < 0)
1903            {
1904              stop += dim_size;
1905              if (stop < lower) stop = lower;
1906            }
1907            else if (stop > upper)
1908            {
1909              stop = upper;
1910            }
1911          }
1912
1913          ptrdiff_t len;
1914          if (step > 0)
1915          {
1916            len = (start < stop) ? (stop - start - 1_z) / step + 1_z : 0_z;
1917          }
1918          else
1919          {
1920            len = (start > stop) ? (start - stop - 1_z) / (-step) + 1_z : 0_z;
1921          }
1922
1923          ret.m_shape[i] = static_cast<size_t>(len);
1924          ret.m_offset += start * ret.m_strides[i];
1925          ret.m_strides[i] *= step;
1926        }
1927        // For SliceAll, don't modify shape
1928      }, slice);
1929      ++i;
1930    }
1931
