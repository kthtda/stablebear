20  /// A point cloud of shape (n_points, dim).
21  ///
22  /// A PointCloud either owns its coordinates or is an indexed view: it shares
23  /// another cloud's coordinate buffer and selects points through an attached
24  /// index set. Access via n_points()/dim()/operator()(i, j) is transparent to
25  /// which mode it is in, so consumers need no special case. This
26  /// lets a tensor of subsamples store one shared source plus small index arrays
27  /// instead of re-storing every (possibly high-dimensional) point.
28  ///
29  /// Deliberately not a Tensor<T>: the raw coordinate storage and the selected
30  /// points disagree for indexed views, so tensor-level access has no single
31  /// meaning here. Use the cloud-level members for the selected points, or
32  /// coords() to reach the underlying storage explicitly.

170      for (size_t i = 0; i < n; ++i)
171      {
172        for (size_t j = 0; j < d; ++j)
173        {
174          if ((*this)(i, j) != rhs(i, j))
175          {
176            return false;
177          }
178        }
179      }
180      return true;
181    }
182
183    /// Deep copy. Tensor cells route stores through detail::store_copy, which
184    /// prefers copy(). An owning cloud copies its coordinates. An indexed view
185    /// copies its index array (so cells don't alias) and, by default
186    /// (@p keepSource), keeps sharing the source coordinates — immutable by
187    /// convention, the point of indexed views; with @p keepSource false it also
188    /// deep-copies the source, yielding a view that aliases nothing.
189    /// @p keepSource is moot for an owning cloud, which never shares.
190    [[nodiscard]] PointCloud copy(bool keepSource = true) const
191    {
192      if (m_coords.rank() == 0)
193      {
194        return PointCloud();
195      }
196      if (is_indexed())
197      {
198        if (keepSource)
199        {
200          return PointCloud(m_coords, m_indices.copy());
201        }
202        return PointCloud(m_coords.copy(), m_indices.copy());
203      }
204      return PointCloud(m_coords.copy());
205    }
206
207    /// Validate a point selection without constructing the selected cloud.
208    void validate_index(const index_type& selection) const
209    {
210      if (m_coords.rank() != 2)
211      {
212        throw std::invalid_argument("Point-cloud coordinates must have 2 dimensions");
213      }
214
215      const Tensor<uint64_t>& requested = selection;
216      if (requested.rank() != 1)
217      {
218        throw std::invalid_argument(
219          "Point-cloud selections must have 1 dimension, got "
220          + std::to_string(requested.rank()));
221      }
222
223      for (size_t i = 0; i < requested.shape(0); ++i)
224      {
225        if (requested(i) >= n_points())
226        {
227          throw std::out_of_range(
228            "Point-cloud index " + std::to_string(requested(i))
229            + " is out of bounds for a cloud with "
230            + std::to_string(n_points()) + " points");
231        }
232      }
233    }
234
235    /// Validate and apply a point selection. Indexed tensors also validate at
236    /// construction for immediate errors, but retain this check because their
237    /// shared source elements may subsequently be replaced.
238    [[nodiscard]] PointCloud index_into(const index_type& selection) const
239    {
240      validate_index(selection);
241      const Tensor<uint64_t>& requested = selection;
242
243      if (!is_indexed())
244      {
245        return PointCloud(m_coords, requested);
246      }
247
248      Tensor<uint64_t> resolved({requested.shape(0)});
249      for (size_t i = 0; i < requested.shape(0); ++i)
250      {
251        const uint64_t logicalIndex = requested(i);
252        resolved(i) = m_indices(static_cast<size_t>(logicalIndex));
253      }
254      return PointCloud(m_coords, std::move(resolved));
255    }
256
257    /// Return an owning value containing exactly the logical coordinates.
258    [[nodiscard]] PointCloud materialized_copy() const
259    {
260      if (m_coords.rank() == 0)
261      {
262        return PointCloud();
263      }
264      return is_indexed() ? PointCloud(materialize()) : PointCloud(m_coords.copy());
265    }
266
267    /// Materialize the selected points into a contiguous coordinate tensor.
268    /// Returns the coordinates as-is when not indexed.
269    [[nodiscard]] Tensor<T> materialize() const
270    {
271      if (!is_indexed())
272      {
273        return m_coords;
274      }
275
276      const size_t n = n_points();
277      const size_t d = dim();
278      Tensor<T> out({n, d});
279      for (size_t i = 0; i < n; ++i)
280      {
281        const auto row = static_cast<size_t>(m_indices(i));
282        for (size_t j = 0; j < d; ++j)
283        {
284          out({i, j}) = m_coords({row, j});
285        }
286      }
287      return out;
288    }
289
290    [[nodiscard]] Tensor<T>& mutable_coords()
291    {
292      if (is_indexed())
293      {
294        throw std::logic_error(
295          "Cannot mutate an indexed PointCloud directly; materialize its shared tensor backing first");
296      }
297      return m_coords;
298    }
299
300  private:
301    void validate_coordinates() const
302    {
303      if (m_coords.rank() != 2)
304      {
305        throw std::invalid_argument(
306          "Point-cloud coordinates must have 2 dimensions, got "
307          + std::to_string(m_coords.rank()));
308      }
309    }
310
311    Tensor<T> m_coords;         // (n_source_points, dim), possibly shared
312    Tensor<uint64_t> m_indices; // rank-1 when an indexed view, empty otherwise
313  };

315  /// Identifies PointCloud<T> instantiations (exposing scalar_type = T), for
316  /// the io and Python binding layers.
317  template <typename T>
318  struct is_point_cloud : std::false_type {};
319
320  template <typename T>
321  struct is_point_cloud<PointCloud<T>> : std::true_type { using scalar_type = T; };
322
323  template <typename T>
324  inline constexpr bool is_point_cloud_v = is_point_cloud<T>::value;

326  /**
327   * Cast a tensor of point clouds (Tensor<PointCloud<U>>) to a different precision
328   * (Tensor<PointCloud<T>>), converting each point cloud's coordinates.
329   *
330   * Sharing carries over: each distinct coordinate buffer is cast once
331   * (deduplicated by storage identity, as in the io layer) and indexed views are
332   * rebuilt on top of the cast source with their own index arrays. Casting cell
333   * by cell through materialize() would instead expand every view into a full
334   * private copy of its source -- exactly the storage blow-up that indexed
335   * subsample tensors exist to avoid.
336   */
337  template <typename T, typename U>
338  requires std::is_constructible_v<T, U>
339  [[nodiscard]] Tensor<PointCloud<T>> pcloud_cast(const Tensor<PointCloud<U>>& src)
340  {
341    Tensor<PointCloud<T>> result(src.shape());
342
343    // Cast each distinct source buffer once...
344    std::map<std::shared_ptr<const void>, Tensor<T>, std::owner_less<std::shared_ptr<const void>>> castSources;
345    walk(src, [&](const std::vector<size_t>& idx) {
346      const Tensor<U>& coords = src(idx).coords();
347      if (coords.rank() == 0)
348      {
349        return;
350      }
351      if (!castSources.contains(coords.storage_owner()))
352      {
353        castSources.emplace(coords.storage_owner(), tensor_cast<T>(coords));
354      }
355    });
356
357    // ...then rebuild every cell on its shared cast source.
358    walk(src, [&](const std::vector<size_t>& idx) {
359      const PointCloud<U>& cloud = src(idx);
360      if (cloud.coords().rank() == 0)
361      {
362        result(idx) = PointCloud<T>();
363        return;
364      }
365      const Tensor<T>& source = castSources.at(cloud.coords().storage_owner());
366      if (cloud.is_indexed())
367      {
368        result(idx) = PointCloud<T>(source, cloud.indices().copy());
369      }
370      else
371      {
372        result(idx) = PointCloud<T>(source);
373      }
374    });
375    return result;
376  }
