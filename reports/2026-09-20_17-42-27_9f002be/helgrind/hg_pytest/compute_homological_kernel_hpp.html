32    // Performs the cross filtration: replays the d' merges (the births w_i) and
33    // computes each death v_i against d.
34    //
35    // The death of merge i is the scale at which [a_i] and [b_i] connect in the
36    // quotient of (X, d) by the i-1 earlier contractions — the subdominant ultra
37    // pseudo-metric of the quotient space. A contraction can create a shortcut
38    // through the contracted component that lowers the connection scale of
39    // other pairs, so each death depends on the whole set of earlier
40    // contractions, not just the merging pair.
41    //
42    // Connectivity at scale t of (X, d) equals connectivity of the d-MST
43    // restricted to edges <= t, and contractions only add identifications on
44    // top. So each death is answered by a Kruskal sweep: seed a union-find with
45    // the current d'-components, add d-MST edges in ascending order, and stop
46    // as soon as a_i ~ b_i; the weight of the connecting edge is the death.
47    //
48    // The sweep for merge i stops after rank(v_i) edges, and the deaths are a
49    // permutation of the d-MST edge ranks, so all sweeps together perform
50    // exactly n(n-1)/2 bounded union-find operations plus an O(n) seed copy per
51    // merge: O(n^2 alpha(n)) time, O(n) memory. Throws if some death lands
52    // below its birth (beyond roundoff), which means d does not dominate d'.
53    template <typename T>
54    void compute_cross_filtration(
55        const std::vector<MergeEdge<T>> &primeMerges, const std::vector<MergeEdge<T>> &dMerges,
56        std::vector<PersistencePair<T>> &bars)
57    {
58      bars.clear();
59      if (primeMerges.empty())
60      {
61        return;
62      }
63      // Exactly n-1 merges for n points is an invariant the callers establish
64      // (both merge vectors come from mst_merge_order), so the point count is
65      // recoverable from the merge count.
66      const size_t n = primeMerges.size() + 1;
67      bars.reserve(primeMerges.size());
68
69      // primeComponents holds the components induced by the contractions
70      // performed so far (= the current d'-components); dComponents is the
71      // working copy each death query runs on, adding d-MST edges on top of
72      // those contractions. The copy assignment reuses capacity, so the
73      // steady state allocates nothing.
74      UnionFind primeComponents(n);
75      UnionFind dComponents(n);
76
77      for (auto const &merge : primeMerges)
78      {
79        dComponents = primeComponents;
80
81        // The d'-components alone can never connect merge.a to merge.b (a and
82        // b are in different ones by definition of this merge), and the full
83        // d-MST is spanning — so the loop always runs at least once and exits
84        // with edgeIdx <= n-1, having set death.
85        //
86        // The connectivity check is find-free: rootA/rootB are cached, and a
87        // cached root can only stop being one when a union absorbs it
88        // (absorbedRoot below) — so patching on that case keeps both caches
89        // exact.
90        auto rootA = dComponents.find(merge.a);
91        auto rootB = dComponents.find(merge.b);
92        auto death = std::numeric_limits<T>::quiet_NaN();
93        size_t edgeIdx = 0;
94        while (rootA != rootB)
95        {
96          auto const &edge = dMerges[edgeIdx++];
97          // Unconditional union: an edge whose endpoints are already connected
98          // (through the d'-contractions or earlier edges) is a harmless
99          // self-union.
100          auto survivingRoot = dComponents.find(edge.a);
101          auto absorbedRoot = dComponents.find(edge.b);
102          dComponents.unite(survivingRoot, absorbedRoot);
103          if (absorbedRoot == rootA)
104          {
105            rootA = survivingRoot;
106          }
107          else if (absorbedRoot == rootB)
108          {
109            rootB = survivingRoot;
110          }
111          death = edge.mergeDist;
112        }
113
114        // Births (from d') and deaths (from d) can reach mathematically equal
115        // values through different arithmetic, and inputs materialized at large
116        // coordinate magnitude (e.g. projected point clouds) carry absolute
117        // error ~eps * |coordinate|, which can dwarf a few ULPs of the
118        // distances. sqrt(eps) relative forgives any plausible roundoff, while
119        // genuine domination failures — which violate at data scale — still
120        // throw.
121        const auto tolerance =
122            std::sqrt(std::numeric_limits<T>::epsilon()) * std::max(std::abs(merge.mergeDist), std::abs(death));
123        if (death < merge.mergeDist - tolerance)
124        {
125          throw std::runtime_error("homological kernel: d does not dominate d' (death below birth)");
126        }
127        bars.emplace_back(merge.mergeDist, std::max(death, merge.mergeDist));
128
129        // Only now does the contraction enter primeComponents: merge i must
130        // see the quotient by merges 1..i-1, not by itself or later ones.
131        primeComponents.unite(primeComponents.find(merge.a), primeComponents.find(merge.b));
132      }
133    }

135    // Full kernel computation for one instance: d' merge order (the births),
136    // d-MST (the death timeline), cross filtration. Requires d' <= d pointwise;
137    // compute_cross_filtration throws otherwise. The instance size comes from
138    // dDist.size(); callers validate that both oracles agree on it. For
139    // sizes <= 1 the kernel barcode is empty.
140    // mergeDistTransform maps each stored merge distance from the scale the
141    // DistT functor computes in to the scale of the output bars (sqrt for the
142    // squared-Euclidean oracle, identity for oracles already in bar scale); it
143    // must be monotone so the merge order is unchanged.

146    void homological_kernel_single_impl(
147        const DistT &dDist, const DistT &dPrimeDist, Barcode<T> &ret, PostF mergeDistTransform = {})
148    {
149      std::vector<MergeEdge<T>> primeMerges; // d' merge order: the births
150      std::vector<MergeEdge<T>> dMerges;     // d-MST edges: the sweep timeline that answers the deaths
151      mst_merge_order(dPrimeDist, primeMerges);
152      mst_merge_order(dDist, dMerges);
153      for (auto &m : primeMerges)
154      {
155        m.mergeDist = mergeDistTransform(m.mergeDist);
156      }
157      for (auto &m : dMerges)
158      {
159        m.mergeDist = mergeDistTransform(m.mergeDist);
160      }
161
162      std::vector<PersistencePair<T>> bars;
163      compute_cross_filtration(primeMerges, dMerges, bars);
164      ret = std::move(bars);
165    }
