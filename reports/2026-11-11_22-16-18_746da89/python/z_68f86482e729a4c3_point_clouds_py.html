28def to_point_clouds(series: TimeSeries | TimeSeriesTensor | _PandasLike, times: Any = None, *, 

29 n_points: int | None = None, 

30 step: float | _ScalarDuration | None = None) -> PointCloud | PointCloudTensor: 

31 """The point clouds traced by a time series or by each series of a 

32 tensor of them, such as windows. 

33 

34 The observation in effect at each of a set of times in the 

35 :attr:`~TimeSeries.domain` gives one point, in the order of the times: 

36 an observation of shape ``()`` a point of dimension 1, and one of shape 

37 ``(m,)`` a point of dimension ``m``. The times are, by default, those 

38 at which the series stores an observation. A window 

39 (:func:`cut_windows`) stores the value in effect at its start at time 

40 0, so that value is a point if the window includes its start, and is 

41 left out if the start is open. 

42 

43 With ``n_points``, the times are ``n_points`` evenly spaced times over 

44 the domain of each series, including each end the domain includes: 

45 with both ends closed they are ``length / (n_points - 1)`` apart, with 

46 one end open ``length / n_points`` apart from the closed end, and with 

47 both open ``length / (n_points + 1)`` apart; one point lies at the 

48 closed end, or in the middle if both are open. These are the evenly 

49 spaced times over the domain with both ends closed, with 

50 ``n_points + 1`` or ``n_points + 2`` times when ends are open and the 

51 open ends left out. The times thus depend 

52 on which ends are closed: ``[0, 2]`` gives times 0, 1, 2, and 

53 ``[0, 2)`` gives 0, 2/3, 4/3. The clouds of half-open windows that 

54 follow each other then form one evenly spaced grid, without a time 

55 in two clouds. Windows of one size give clouds of ``n_points`` points 

56 each. 

57 With ``step``, the times are ``start + k * step`` in the domain of each 

58 series, for ``k = 0, 1, ...``, leaving out ``start`` if the domain 

59 leaves it out. With ``times``, the times are those, in their order, the 

60 same for every series, or for dates and times the same instants, so the 

61 cloud holds the values that evaluating the series at ``times`` gives. 

62 

63 A ``float32`` series gives ``pcloud32`` clouds, and series of other 

64 numbers ``pcloud64`` clouds. A pandas ``Series`` or ``DataFrame`` is 

65 converted with :meth:`TimeSeries.from_pandas` first. 

66 

67 Parameters 

68 ---------- 

69 series : TimeSeries, TimeSeriesTensor, pandas.Series, or pandas.DataFrame 

70 Series of one channel of numbers. Combine several channels with 

71 :func:`~stablebear.tensor_create.stack` or 

72 :func:`~stablebear.tensor_create.concatenate`, or select one with 

73 ``series[i]``. 

74 times : TimeDeltaTensor, DateTimeTensor, or array_like, optional 

75 The times of the points, counted from the time 0 of each series, 

76 as :meth:`TimeSeries.__call__` takes them: durations, such as a 

77 :class:`TimeDeltaTensor`, for series with a unit, or numbers for 

78 series without one. Series with an :attr:`~TimeSeries.epoch` also 

79 take dates and times, such as a :class:`DateTimeTensor`, which each 

80 series places at ``date - epoch - origin`` in its own time. 

81 n_points : int, optional 

82 The number of evenly spaced points of each cloud. 

83 step : float, TimeDelta, tuple, numpy.timedelta64, datetime.timedelta, or pandas.Timedelta, optional 

84 The time between points: a duration for a series with a unit, or a 

85 number for one without, as :func:`cut_windows` takes its ``step``. 

86 

87 Returns 

88 ------- 

89 PointCloud or PointCloudTensor 

90 The cloud of a series, or for a tensor of series a tensor of the 

91 same shape holding the cloud of each, whose clouds can differ in 

92 size. 

93 

94 Raises 

95 ------ 

96 TimeSeriesDomainError 

97 If a time is outside the domain of a series, with the ``index`` 

98 of the first such series of a tensor. 

99 ValueError 

100 If a series has several channels, more than one of ``times``, 

101 ``n_points``, and ``step`` is given, ``n_points`` is not positive or 

102 is 1 for a domain with both ends closed, ``step`` is not finite and positive, 

103 ``times`` is not one-dimensional, or for dates and times on a 

104 series without an epoch. 

105 TypeError 

106 If the observations are not numbers, are booleans, or have more 

107 than one dimension, for a ``step`` or ``times`` that are numbers 

108 on a series with a unit or durations on one without, or for an 

109 object that is not a series or a pandas ``Series`` or 

110 ``DataFrame``. 

111 """ 

112 if isinstance(series, TimeSeriesTensor): 

113 _check_numbers(series.dtype.value_type) 

114 n_points, step, times, dates, unit = _point_times(times, n_points, step, None) 

115 counts = None 

116 if dates is not None: 

117 counts, unit = _date_counts(dates) 

118 counts = np.asarray(counts).tolist() 

119 try: 

120 return PointCloudTensor(cpp.to_point_clouds(series._data, n_points, step, times, counts, unit)) 

121 except cpp.TimeSeriesDomainError as error: 

122 if dates is None: 

123 raise series._domain_error(error) from None 

124 raise series._naming_dates(error, dates) from None 

125 series = _as_series(series) 

126 _check_numbers(series.dtype)

128 n_points, step, times, _, unit = _point_times(times, n_points, step, series) 

129 try: 

130 return PointCloud(cpp.to_point_cloud(series._data, n_points, step, times, None, unit)) 

131 except cpp.TimeSeriesDomainError as error: 

132 raise _domain_error(error, series.unit, None) from None

140def _point_times(times, n_points, step, series): 

141 """The ``n_points``, ``step``, ``times``, ``dates``, and ``unit`` 

142 arguments of the C++ ``to_point_clouds``, 0 or ``None`` for a mode not chosen. Dates 

143 and times are placed in ``series`` here, or for a tensor of series, with 

144 ``series`` ``None``, returned as a ``datetime64`` array, which the caller 

145 passes as counts for C++ to place in each series.""" 

146 if sum(x is not None for x in (times, n_points, step)) > 1: 

147 raise ValueError("Pass at most one of times, n_points, and step") 

148 if times is not None: 

149 # A DateTimeTensor is read as dates before _queries, which would 

150 # convert it one element at a time. 

151 if isinstance(times, DateTimeTensor): 

152 numbers, unit = times.to_datetime64(), None 

153 else: 

154 numbers, unit = _queries(times) 

155 if isinstance(numbers, np.ndarray) and numbers.dtype.kind in "MOUS": 

156 if series is None: 

157 dates = _checked_dates(_as_dates(numbers)) 

158 if dates.ndim != 1: 

159 raise ValueError(f"The times of the points must be one-dimensional, not of shape {dates.shape}") 

160 return 0, 0.0, None, dates, None 

161 numbers, unit = _times_of_dates(numbers, series) 

162 numbers = np.asarray(numbers, dtype=np.float64) 

163 if numbers.ndim != 1: 

164 raise ValueError(f"The times of the points must be one-dimensional, not of shape {numbers.shape}") 

165 return 0, 0.0, numbers.tolist(), None, unit 

166 if n_points is not None: 

167 n_points = operator.index(n_points) 

168 if n_points < 1: 

169 raise ValueError(f"n_points must be positive, not {n_points}") 

170 return n_points, 0.0, None, None, None 

171 if step is not None: 

172 step = _duration(step, "The step must be finite and positive") 

173 if isinstance(step, TimeDelta): 

174 return 0, _in_unit(step, step.unit), None, None, step.unit 

175 return 0, step, None, None, None 

176 return 0, 0.0, None, None, None

179def _times_of_dates(dates, series): 

180 """``dates``, an array of dates and times, as times of ``series`` and 

181 its unit.""" 

182 if series.epoch is None: 

183 raise ValueError("This series has no epoch to place dates and times; set one with with_epoch") 

184 dates = _checked_dates(_as_dates(dates)) 

185 times = series._time_of_date(dates, 0) 

186 series._check_date_in_domain(dates, times, 0) 

187 return times, series.unit
