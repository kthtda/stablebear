94def zeros(shape: ShapeLike, dtype: Dtype = pcf32) -> Any: # Any: the type follows the dtype. 

95 """ 

96 Creates a new `Tensor` of the specified `shape` and `dtype` whose entries are "zero." What "zero" means depends on the `dtype`: 

97 

98 `dtype=pcf32/64`: A PCF that takes the value 0 for all times. 

99 `dtype=pcf32i/64i`: An integer PCF that takes the value 0 for all times. 

100 `dtype=float32/float64`: The number 0. 

101 `dtype=pcloud32/64`: An empty point cloud. 

102 `dtype=barcode32/64`: An empty barcode. 

103 `dtype=symmat32/64`: A 0×0 symmetric matrix. 

104 `dtype=distmat32/64`: A 0×0 distance matrix. 

105 `dtype=ts(value_type)`: A series of one channel without observations, 

106 in seconds, to be replaced by assigning a `TimeSeries` to the element. 

107 Evaluating a series without observations raises `ValueError`. 

108 

109 Parameters 

110 ---------- 

111 shape : ShapeLike 

112 Shape of the returned tensor 

113 dtype 

114 The data type of the elements 

115 

116 Returns 

117 ------- 

118 Tensor 

119 The newly created tensor 

120 """

122 if not isinstance(shape, Shape): 

123 shape = Shape(shape) # If passed as, e.g., tuple of ints 

124 

125 if isinstance(dtype, _SeriesDtype): 

126 from .timeseries.series import TimeSeriesTensor 

127 return TimeSeriesTensor(getattr(cpp, f"TimeSeries{_NATIVE_SUFFIX[dtype.value_type]}Tensor")(shape))

135def tensor(data, dtype: Dtype | None = None) -> Any: # Any: the type follows the data and dtype. 

136 """Create a tensor from existing data, dispatching on *dtype*. 

137 

138 A NumPy-like factory that wraps the per-family constructors so any 

139 supported tensor can be built from an ndarray (or nested list) in one 

140 call, instead of allocating with :func:`zeros` and assigning elements in 

141 a Python loop. 

142 

143 Parameters 

144 ---------- 

145 data : ndarray, list, or tuple 

146 The source data. For point-cloud / matrix / barcode tensors this is 

147 interpreted by the corresponding constructor (see those classes). 

148 dtype : Dtype, optional 

149 Target element dtype. When ``None``, a numeric dtype is inferred from 

150 the array (bool/int/float), while a collection of ``Pcf`` objects is 

151 inferred as a ``PcfTensor`` or ``IntPcfTensor``, and one of 

152 :class:`~stablebear.timeseries.TimeSeries` as a 

153 :class:`~stablebear.timeseries.TimeSeriesTensor` of dtype 

154 ``ts(value_type)``. The series may differ in their times, units, 

155 and channels, but not in the dtype of their observations. Other 

156 non-numeric tensors require an explicit dtype. 

157 

158 Raises 

159 ------ 

160 TypeError 

161 If series differ in the dtype of their observations, or do not 

162 match an explicit ``dtype``. 

163 

164 Returns 

165 ------- 

166 Tensor 

167 """ 

168 def contains_tensor(value): 

169 if isinstance(value, Tensor): 

170 return True 

171 return isinstance(value, (list, tuple)) and any( 

172 contains_tensor(child) for child in value 

173 ) 

174 

175 if isinstance(data, (list, tuple)) and contains_tensor(data): 

176 return NestedTensor(data, dtype=dtype) 

177 

178 from .timeseries.series import TimeSeries, TimeSeriesTensor 

179 if isinstance(dtype, _SeriesDtype) or (dtype is None and isinstance(_first_leaf(data), TimeSeries)): 

180 return TimeSeriesTensor._from_series(data, dtype) 

181 

182 if dtype is None: 

183 if _first_pcf(data) is not None: 

184 return _to_tensor_pcf(data) 

185 

186 import numpy as np 

187 arr = np.asarray(data) 

188 if arr.dtype == np.bool_: 

189 return BoolTensor(arr) 

190 if np.issubdtype(arr.dtype, np.integer): 

191 return IntTensor(arr) 

192 if np.issubdtype(arr.dtype, np.floating): 

193 return FloatTensor(arr) 

194 raise TypeError( 

195 f"Cannot infer a tensor dtype from data of dtype {arr.dtype}; " 

196 "pass an explicit dtype= argument")

201 raise TypeError(f"Unsupported dtype {dtype} for tensor()") 

202 wrapper = dispatch[dtype][0] 

203 if dtype in _DATA_DTYPE_KW: 

204 return wrapper(data, dtype=dtype) 

205 return wrapper(data)

242def concatenate(tensors, axis=0): 

243 """Concatenate tensors, or the observations of time series, along an 

244 existing axis. 

245 

246 Tensors are joined along their axis ``axis`` (outer indexing): tensors 

247 of shapes ``(2, 4)`` and ``(3, 4)`` give one of shape ``(5, 4)``. 

248 

249 For time series, the result is a series of one channel whose 

250 observation at each time joins the channels' observations in effect 

251 there along their axis ``axis``: channels of shapes ``(2,)`` and 

252 ``(3,)`` give observations of shape ``(5,)``. 

253 Its times are the start of the domain and every time in the domain at 

254 which a channel stores an observation. The domain, unit, epoch, and 

255 origin stay the same, and the snap tolerance is the smallest of the 

256 channels'. Observations are copied. 

257 

258 .. warning:: 

259 

260 Observations can be dropped or repeated. A channel's observations 

261 outside the domain, or at an end the domain leaves out, are 

262 dropped, and an observation in effect at several of the times is 

263 repeated at each. 

264 

265 Parameters 

266 ---------- 

267 tensors : sequence of Tensor, or TimeSeries 

268 Tensors of one type, whose shapes match except along ``axis``. A 

269 ``TimeSeries`` joins its channels; make several series the 

270 channels of one with 

271 :meth:`~stablebear.timeseries.TimeSeries.from_channels` first. 

272 To put separate series into a tensor, use :func:`tensor`. 

273 axis : int, default 0 

274 The axis to join along, of each tensor or of each observation; a 

275 negative ``axis`` counts from the end. 

276 

277 Returns 

278 ------- 

279 Tensor or TimeSeries 

280 A tensor of the type of the inputs, or a series of one channel. 

281 

282 Raises 

283 ------ 

284 ValueError 

285 If there is nothing to join, the shapes differ other than along 

286 ``axis``, or ``axis`` is out of range. For series, also if the 

287 observations are scalars, the channels have different origins, 

288 their domains do not overlap, or a channel has no observations. 

289 TypeError 

290 If a sequence holds a ``TimeSeries``. 

291 """ 

292 series = _series_channels(tensors) 

293 if series is not None: 

294 return series._from_cpp(cpp.concatenate_channels(series._data, operator.index(axis))) 

295 if not tensors: 

296 raise ValueError("need at least one tensor to concatenate") 

297 _validate_nested_join(tensors) 

298 cpp_tensors = [t._data for t in tensors] 

299 result = type(cpp_tensors[0]).concatenate(cpp_tensors, axis) 

300 return tensors[0]._to_py_tensor(result)

383def split(tensor, indices_or_sections, axis=0): 

384 """Split a tensor into sub-tensors along an axis. 

385 

386 Parameters 

387 ---------- 

388 tensor : Tensor 

389 The tensor to split. 

390 indices_or_sections : int or list of int 

391 If an int, the axis length must be evenly divisible by that number. 

392 The result contains that many contiguous parts, each of length 

393 ``axis_size // sections``. Use ``array_split`` when equal division is 

394 not possible. 

395 If a list ``[i0, i1, ...]``, its entries are boundaries and the 

396 resulting parts cover the half-open intervals ``[0, i0)``, 

397 ``[i0, i1)``, ..., ``[ik, axis_size)`` along *axis*. For example, 

398 splitting an axis of length 8 at ``[3, 5]`` produces parts of lengths 

399 3, 2, and 3. The element at each boundary starts the following part. 

400 axis : int 

401 The axis along which to split (default 0). A negative axis counts 

402 from the end. 

403 

404 Returns 

405 ------- 

406 list of Tensor 

407 A list of tensor views sharing data with the original. 

408 

409 See Also 

410 -------- 

411 array_split : Split allowing uneven divisions. 

412 """ 

413 cpp_data = tensor._data 

414 if isinstance(indices_or_sections, int): 

415 parts = type(cpp_data).split_sections(cpp_data, indices_or_sections, axis) 

416 else: 

417 parts = type(cpp_data).split_indices(cpp_data, list(indices_or_sections), axis) 

418 return [tensor._to_py_tensor(p) for p in parts]
