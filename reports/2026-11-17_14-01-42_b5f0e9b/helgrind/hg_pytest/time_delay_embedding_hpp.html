24  /// The delay embedding of @p channel with @p dimension components spaced
25  /// @p delay apart, in the channel's unit, with w = (dimension - 1) * delay.
26  /// Its value at t is (x(t - w), ..., x(t - delay), x(t)), oldest first as
27  /// in Kantz and Schreiber's convention, for every t from t_0 + w to the end
28  /// of the channel's domain. If @p forward, it is instead
29  /// (x(t), x(t + delay), ..., x(t + w)), also oldest first, for every t from
30  /// t_0 to the end of the domain minus w. Each end is included when the
31  /// channel's is, and each observation gains a leading axis of length
32  /// @p dimension. Its times are the start of its domain and the times in it
33  /// where a component reaches an observation, t_i + k * delay, or
34  /// t_i - k * delay if @p forward; times that round to the same point are
35  /// recorded once, at the time they snap to. It is a view of the channel's
36  /// storage.

40    inline void check_embedding_arguments(size_t dimension, float64_t delay)
41    {
42      if (dimension == 0)

45      }
46      if (std::isnan(delay) || std::isinf(delay) || delay <= 0)
47      {
48        throw std::invalid_argument("The embedding delay must be finite and positive");

52    /// The lag of each component of an embedding, oldest first: component j
53    /// reads the channel lags[j] before t. Looking forward, the lags are
54    /// negative: component j reads j * delay after t.
55    [[nodiscard]] inline std::vector<float64_t> embedding_lags(size_t dimension, float64_t delay, bool forward)
56    {
57      std::vector<float64_t> lags(dimension);
58      for (size_t j = 0; j < dimension; ++j)
59      {
60        lags[j] = forward ? -(static_cast<float64_t>(j) * delay) : static_cast<float64_t>(dimension - 1 - j) * delay;
61      }
62      return lags;
63    }

65    /// The first and last times of the embedding of @p channel, which has
66    /// observations, with lags from @p lowest to @p highest. A channel with
67    /// an end, such as a window, can be evaluated past its last timestamp up
68    /// to its end, so the embedding can too, up to that end plus the lowest
69    /// lag. Throws if no time has all of the components.
70    template <typename T>
71    std::pair<float64_t, float64_t> embedding_bounds(const Channel<T>& channel, float64_t lowest, float64_t highest)
72    {
73      const auto start = channel.time(0) + highest;
74      const auto end = channel.end().value_or(channel.time(channel.size() - 1)) + lowest;
75      if (channel.end_closed() ? start > end : start >= end)
76      {
77        throw std::invalid_argument("The series is too short for this embedding: no time has all of its components");
78      }
79      return {start, end};
80    }

115      // Component j changes where its lagged time reaches an observation: at

140      // never decreases with time, so a run of the same point ends where some

279  /// such series, an error of the same type whose message names its
280  /// position.
281  template <typename T>
282  std::unique_ptr<DelayEmbedTask<T>> make_delay_embed_task(const Tensor<TimeSeries<T>>& seriesTensor,
283                                                           size_t dimension, float64_t delay, TimeUnit delayUnit,
284                                                           bool forward = false)
285  {
286    detail::check_embedding_arguments(dimension, delay);
287    const DelayEmbedOperation<T> op{.dimension = dimension, .delay = delay, .delayUnit = delayUnit,
288                                    .forward = forward};
289    walk(seriesTensor, [&](const std::vector<size_t>& index) {
290      const auto& series = seriesTensor(index);
291      try
292      {
293        detail::check_embedding_fits(series, dimension, op.delay_in(series), forward);
294      }
295      catch (const TypeError& e)

297        throw TypeError(detail::position_text(index) + ": " + e.what());

301        throw std::invalid_argument(detail::position_text(index) + ": " + e.what());
