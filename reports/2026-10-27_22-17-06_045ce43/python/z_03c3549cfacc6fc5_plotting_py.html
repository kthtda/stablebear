85def plot_distance_weight_heatmap( 

86 distribution, query=(0.0, 0.0), extent=None, 

87 resolution=512, ax=None, **kwargs, 

88): 

89 """Plot distribution weights applied to distance from a query point. 

90 

91 Parameters 

92 ---------- 

93 distribution : Distribution 

94 A distribution from :mod:`stablebear.distributions`. 

95 query : array_like, optional 

96 Two finite coordinates ``(x, y)``, by default the origin. 

97 extent : array_like, optional 

98 Plot bounds ``(xmin, xmax, ymin, ymax)``. If omitted, uses 

99 ``distribution.plot_range()`` to choose a square centered on ``query``: 

100 its half-width is the largest absolute endpoint plus a 10% margin. 

101 Bounds must be finite and strictly increasing on each axis. 

102 resolution : int, optional 

103 Positive number of pixels on each axis, by default 512. Weights are 

104 evaluated at pixel centers within ``extent``. 

105 ax : matplotlib.axes.Axes, optional 

106 Axes to draw on. If omitted, uses the current axes. 

107 **kwargs 

108 Styling arguments passed to ``Axes.imshow``, such as ``cmap``, 

109 ``alpha``, ``norm``, ``vmin``, and ``vmax``. Defaults are 

110 ``cmap="viridis"``, ``aspect="equal"``, and ``interpolation="nearest"``. 

111 The image origin is fixed to ``"lower"``. 

112 

113 Returns 

114 ------- 

115 matplotlib.image.AxesImage 

116 The heatmap, suitable for passing to ``fig.colorbar``. 

117 

118 Notes 

119 ----- 

120 Colors represent ``distribution.weight(distance)``. No normalization over 

121 the grid is performed, and the image is not a probability density over the 

122 plane. This function adds no colorbar, labels, or query marker. 

123 """ 

124 if not isinstance(distribution, _distributions.Distribution): 

125 raise TypeError("distribution must be a Distribution") 

126 resolution = _positive_integer(resolution, "resolution") 

127 query = _finite_vector(query, 2, "query must contain two finite real coordinates") 

128 if extent is None: 

129 radius = 1.1 * max(abs(bound) for bound in distribution.plot_range()) 

130 if not np.isfinite(radius) or radius <= 0: 

131 raise ValueError("cannot choose a finite plot region; pass extent explicitly") 

132 extent = (query[0] - radius, query[0] + radius, 

133 query[1] - radius, query[1] + radius) 

134 extent = _finite_vector(extent, 4, "extent must contain four finite real bounds") 

135 xmin, xmax, ymin, ymax = extent 

136 if xmin >= xmax or ymin >= ymax: 

137 raise ValueError("extent must satisfy xmin < xmax and ymin < ymax") 

138 

139 fractions = (np.arange(resolution) + 0.5) / resolution 

140 x = (1 - fractions) * xmin + fractions * xmax 

141 y = (1 - fractions) * ymin + fractions * ymax 

142 distances = np.hypot(x[None, :] - query[0], y[:, None] - query[1]) 

143 weights = distribution.weight(distances) 

144 

145 ax = plt.gca() if ax is None else ax 

146 kwargs.setdefault("cmap", "viridis") 

147 kwargs.setdefault("aspect", "equal") 

148 kwargs.setdefault("interpolation", "nearest") 

149 return ax.imshow(weights, origin="lower", extent=extent, **kwargs)

152def _resolve_barcode_colors(n, color=None, ax=None): 

153 """Resolve a color specification into one color per barcode. 

154 

155 Parameters 

156 ---------- 

157 n : int 

158 Number of barcodes to plot. 

159 color : color-like object, list of color-like objects or None, optional 

160 ``None`` takes the next ``n`` colors from the axes' property cycle, a 

161 single color is repeated for all barcodes, and a sequence of colors is 

162 cycled across barcodes. 

163 ax : matplotlib axes, optional 

164 Axes whose property cycle supplies default colors. Only used when 

165 ``color`` is ``None``; defaults to the current axes. 

166 

167 Returns 

168 ------- 

169 list of color-like objects 

170 One color per barcode to be plotted. 

171 

172 Raises 

173 ------ 

174 ValueError 

175 If color is neither None, a color-like object, nor a list of color-like objects. 

176 """ 

177 from matplotlib.colors import is_color_like 

178 

179 if color is None: 

180 # Advance the axes' own cycle, as ax.plot does, so set_prop_cycle and 

181 # repeated calls on the same axes behave as in matplotlib. 

182 ax = plt.gca() if ax is None else ax 

183 return [ax._get_lines.get_next_color() for _ in range(n)] 

184 if is_color_like(color): 

185 cycle = [color] 

186 else: 

187 try: 

188 cycle = list(color) 

189 except TypeError: 

190 cycle = [] 

191 if len(cycle) == 0 or not all(is_color_like(c) for c in cycle): 

192 raise ValueError( 

193 f"'color' must be a color or a non-empty sequence of colors; got {color!r}" 

194 ) 

195 return [cycle[i % len(cycle)] for i in range(n)]

198def _infinite_bar_end(bars): 

199 """Return the x value where infinite bars in ``bars`` are drawn to end.""" 

200 deaths = bars[:, 1] 

201 finite = np.isfinite(deaths) 

202 if finite.any(): 

203 return deaths[finite].max() * 1.15 

204 return bars[:, 0].max() + 1.0

207def plot_barcode(bc, ax=None, y_offset=0, **kwargs): 

208 """Plot a persistence barcode as horizontal line segments. 

209 

210 Each bar is drawn as a horizontal segment from birth to death. 

211 Bars with infinite death are drawn as arrows extending to the right 

212 edge of the plot. For a ``BarcodeTensor``, infinite bars in all of its 

213 barcodes end at the same x. 

214 

215 Parameters 

216 ---------- 

217 bc : Barcode, BarcodeTensor or NumPy array 

218 A single ``Barcode``, a 1-D ``BarcodeTensor`` or an (n, 2) NumPy array 

219 of ``(birth, death)`` pairs. For a tensor, 

220 the barcodes are stacked vertically in order. 

221 ax : matplotlib axes, optional 

222 Axes to plot on. If ``None``, uses the current axes. 

223 y_offset : int, optional 

224 Starting y position for the first bar. Useful when stacking 

225 multiple barcodes on the same axes. 

226 **kwargs 

227 Additional keyword arguments passed to 

228 ``matplotlib.collections.LineCollection`` 

229 (e.g. ``linewidth``, ``alpha``, ``label``), except for ``color``, 

230 described below. ``colors`` is not accepted, because the bars are 

231 sorted before drawing, and neither is ``cmap``. 

232 

233 ``color`` is a single color, or a sequence with one color per 

234 barcode that is cycled if shorter than the number of barcodes. 

235 A tuple of three or four numbers is one RGB(A) color, as in 

236 matplotlib. To color barcodes from a colormap, pass its colors, such 

237 as ``mpl.colormaps["viridis"](np.linspace(0, 1, n))`` or 

238 ``mpl.colormaps["tab20"].colors``. A single ``Barcode`` or array is 

239 treated as a one-barcode tensor, so it uses the first color of a 

240 sequence. If ``color`` is omitted, each barcode takes the next color 

241 from the axes' property cycle, continuing across calls on the same 

242 axes. 

243 

244 Returns 

245 ------- 

246 int 

247 The next available y position (for stacking). 

248 

249 Raises 

250 ------ 

251 TypeError 

252 If ``bc`` is not a ``Barcode``, ``BarcodeTensor``, or ``numpy.ndarray``, 

253 if ``bc`` is a ``numpy.ndarray`` containing neither floats nor ints, 

254 or if keyword ``colors`` or ``cmap`` is passed. 

255 ValueError 

256 If ``bc`` is a numpy array that is not of shape ``(n, 2)``, a 

257 ``BarcodeTensor`` with more than one dimension, or if ``color`` is 

258 neither a color nor a non-empty sequence of colors. 

259 """ 

260 from matplotlib.collections import LineCollection 

261 

262 for name in ("colors", "cmap"): 

263 # LineCollection would accept 'colors' (per-segment), but the bars are 

264 # sorted internally so per-segment colors would land in scrambled 

265 # order, and 'cmap' has no effect on it. Only 'color' has 

266 # well-defined semantics here. 

267 if name in kwargs: 

268 raise TypeError( 

269 f"plot_barcode() does not accept '{name}'; use 'color' " 

270 "(a single color, or a sequence with one color per barcode)" 

271 ) 

272 

273 if isinstance(bc, BarcodeTensor): 

274 if len(bc.shape) != 1: 

275 squeezed = bc.squeeze() 

276 if len(squeezed.shape) != 1: 

277 raise ValueError(f"Expected 1-dimensional tensor (got shape {bc.shape})") 

278 return plot_barcode(squeezed, ax=ax, y_offset=y_offset, **kwargs) 

279 colors = _resolve_barcode_colors(bc.shape[0], kwargs.pop("color", None), ax=ax) 

280 # Infinite bars in every barcode end at the same x, computed from all 

281 # bars in the tensor. 

282 all_bars = np.concatenate([np.asarray(bc[i]).reshape(-1, 2) for i in range(bc.shape[0])]) 

283 if len(all_bars) > 0: 

284 kwargs["_inf_end"] = _infinite_bar_end(all_bars) 

285 y = y_offset 

286 for i in range(bc.shape[0]): 

287 y = plot_barcode(bc[i], ax=ax, y_offset=y, color=colors[i], **kwargs) 

288 return y 

289 

290 if not isinstance(bc, (Barcode, np.ndarray)): 

291 raise TypeError(f"Expected Barcode or numpy.ndarray; got {type(bc)}") 

292 

293 # A single barcode behaves like a 1-element tensor for color purposes: 

294 # a sequence of colors resolves to its first color, and no 

295 # color takes the next one from the axes' cycle once the axes exist. 

296 # Explicit colors are validated here, before any axes is created. 

297 color = kwargs.pop("color", None) 

298 if color is not None: 

299 color = _resolve_barcode_colors(1, color)[0] 

300 inf_end = kwargs.pop("_inf_end", None) 

301 

302 if isinstance(bc, np.ndarray): 

303 if bc.ndim != 2 or bc.shape[1] != 2: 

304 raise ValueError(f"Input should have shape (n, 2). Supplied input has shape {bc.shape}") 

305 if not (np.issubdtype(bc.dtype, np.floating) or np.issubdtype(bc.dtype, np.integer)): 

306 raise TypeError(f"Unsupported dtype {bc.dtype}; expected float or integer values") 

307 

308 ax = plt.gca() if ax is None else ax 

309 if color is None: 

310 color = _resolve_barcode_colors(1, ax=ax)[0] 

311 kwargs["color"] = color 

312 

313 bars = np.asarray(bc) 

314 

315 if len(bars) == 0: 

316 return y_offset 

317 

318 # Sort bars lexicographically by (birth, length), descending 

319 lengths = np.where(np.isfinite(bars[:, 1]), bars[:, 1] - bars[:, 0], np.inf) 

320 order = np.lexsort((lengths, bars[:, 0]))[::-1] 

321 bars = bars[order] 

322 

323 defaults = {"linewidth": 1.5} 

324 defaults.update(kwargs) 

325 lw = defaults.get("linewidth", 1.5) 

326 

327 # Infinite bars extend to this point 

328 xmax = _infinite_bar_end(bars) if inf_end is None else inf_end 

329 

330 # Build all bar segments in one pass, using sequential y-positions 

331 segments = [] 

332 inf_y_positions = [] 

333 for i in range(len(bars)): 

334 y = y_offset + i 

335 birth = bars[i, 0] 

336 death = bars[i, 1] 

337 if np.isfinite(death): 

338 segments.append([(birth, y), (death, y)]) 

339 else: 

340 segments.append([(birth, y), (xmax, y)]) 

341 inf_y_positions.append(y) 

342 

343 arrow_color = None 

344 if segments: 

345 lc = LineCollection(segments, **defaults) 

346 ax.add_collection(lc) 

347 arrow_color = tuple(lc.get_colors()[0]) 

348 

349 # Add arrowheads for infinite bars 

350 for y in inf_y_positions: 

351 ax.annotate("", 

352 xy=(xmax, y), 

353 xytext=(xmax - (xmax * 0.02), y), 

354 arrowprops=dict(arrowstyle="-|>", color=arrow_color, lw=lw)) 

355 

356 ax.autoscale_view() 

357 ax.yaxis.set_ticks([]) 

358 

359 return y_offset + len(bars)

413def plot_timeseries(series, ax=None, *, unit=None, relative=False, markers=False, **kwargs): 

414 """Plot a time series as the step function it evaluates to. 

415 

416 Each channel is drawn from its first observation to the end of its 

417 domain, holding each value until the next observation, at the times 

418 where it lies in the series it was cut from: a window is drawn where it 

419 was cut, at its :attr:`~stablebear.timeseries.TimeSeries.origin` plus its 

420 own times, unless ``relative`` is true. 

421 

422 Parameters 

423 ---------- 

424 series : TimeSeries or TimeSeriesTensor 

425 A time series of scalar observations, or a 1-D tensor of them, such 

426 as windows, drawn one line each, each taking the next color of the 

427 axes unless ``color`` is given. A time series with several channels 

428 is drawn with one row per channel, in channel order, on axes that 

429 share the time axis; only the last row is labeled. 

430 ax : matplotlib axes or sequence of them, optional 

431 Axes to draw on. If ``None``, uses the current axes, or for several 

432 channels creates a new figure with one row per channel. For several 

433 channels, a sequence of one axes per channel, used in order and as 

434 given, without making them share x. 

435 unit : str, optional 

436 The time unit of the x axis, one of the units of 

437 :class:`~stablebear.timeseries.Times`. Defaults to the unit the axis 

438 already has, or else the unit of ``series``. 

439 relative : bool, default False 

440 Whether to draw each time series at its own times, from 0, instead 

441 of where it lies in the series it was cut from. Windows are then 

442 drawn on top of each other, to compare their shapes. 

443 markers : bool, default False 

444 Whether to mark each observation with a dot, which also shows a 

445 last value that holds only at the end of the domain. 

446 **kwargs 

447 Passed to ``Axes.plot``, such as ``color``, ``linewidth``, or 

448 ``label``. 

449 

450 Returns 

451 ------- 

452 list of matplotlib.lines.Line2D 

453 The lines drawn. 

454 

455 Raises 

456 ------ 

457 ValueError 

458 If ``unit`` differs from the unit the x axis already has, the 

459 observations are not scalars, or a tensor has more than one 

460 dimension; for a time series with an epoch, if ``unit`` is given 

461 or the x axis has a time unit, unless ``relative`` is true. 

462 TypeError 

463 If the observations are not numbers, such as point clouds or 

464 barcodes, or if the x axis holds dates or other values that are not 

465 times, for a time series drawn at times. 

466 

467 Notes 

468 ----- 

469 A time series with an :attr:`~stablebear.timeseries.TimeSeries.epoch` is 

470 drawn at dates and times, ``epoch + origin + t`` rounded to 

471 microseconds, on a date axis with no label, unless ``relative`` is 

472 true. Otherwise the x axis keeps one time unit, stored on it by the first time series 

473 drawn, and each time series is converted into it exactly, so series in 

474 different units line up. The axis is labeled ``time (<unit>)`` unless it 

475 already has a label. 

476 """ 

477 from .timeseries.series import Times, TimeSeriesTensor 

478 

479 options = dict(unit=unit, relative=relative, markers=markers, **kwargs) 

480 if isinstance(series, TimeSeriesTensor): 

481 tensor = series.squeeze() if len(series.shape) != 1 else series 

482 if len(tensor.shape) != 1: 

483 raise ValueError(f"Expected a 1-D tensor of time series, got shape {series.shape}") 

484 if ax is None and tensor.shape[0] > 0 and tensor[0].n_channels > 1: 

485 ax = _channel_rows(tensor[0].n_channels) 

486 return [line for k in range(tensor.shape[0]) for line in plot_timeseries(tensor[k], ax=ax, **options)] 

487 if series.n_channels > 1: 

488 return _plot_channels(series, _channel_rows(series.n_channels) if ax is None else _rows(ax, series.n_channels), 

489 **options) 

490 if series.dtype not in _NUMERIC_DTYPES: 

491 raise TypeError(f"Can only plot time series of numbers, not of {series.dtype}") 

492 ax = plt.gca() if ax is None else ax 

493 dated = series.epoch is not None and not relative 

494 if dated: 

495 if unit is not None: 

496 raise ValueError("A time series with an epoch is drawn at dates, which have no unit; " 

497 "pass unit only with relative=True") 

498 if isinstance(ax.xaxis.get_units(), str): 

499 raise ValueError(f"The x axis is a time axis in {ax.xaxis.get_units()!r}, not a date axis") 

500 else: 

501 requested = None if unit is None else series.to(unit).unit 

502 series = series.to(_time_axis_unit(ax, requested or series.unit, requested)) 

503 offset = 0.0 if relative else series.origin 

504 times = np.asarray(series.times) + offset 

505 values = np.asarray(series.values) 

506 if values.ndim != 1: 

507 raise ValueError(f"Can only plot scalar observations, not observations of shape {series.shape}") 

508 if markers: 

509 kwargs = {"marker": "o", "markevery": list(range(len(times))), **kwargs} 

510 end = series.domain.end + offset 

511 if end > times[-1]: 

512 # The last value holds until the end of the domain. 

513 times = np.append(times, end) 

514 values = np.append(values, values[-1]) 

515 if dated: 

516 # Dates at epoch + origin + t, at microseconds; the axis is a date axis. 

517 times = Times(times, unit=series.unit, epoch=series.epoch).to_datetime64() 

518 return ax.plot(times, values, drawstyle="steps-post", **kwargs) 

519 lines = ax.plot(times, values, drawstyle="steps-post", **kwargs) 

520 if not ax.get_xlabel(): 

521 ax.set_xlabel(f"time ({series.unit})") 

522 return lines
