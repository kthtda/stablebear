382   * (Tensor<PointCloud<T>>), converting each point cloud's coordinates.
383   *
384   * Dense coordinate sharing carries over: each distinct coordinate buffer is
385   * cast once. A transient indexed view is materialized because an ordinary
386   * tensor must contain self-contained point-cloud values.
387   */
388  template <typename T, typename U>
389  requires std::is_constructible_v<T, U>
390  [[nodiscard]] Tensor<PointCloud<T>> pcloud_cast(const Tensor<PointCloud<U>>& src)
391  {

393
394    // Cast each distinct source buffer once...
395    std::map<const void*, PointCloud<T>> castSources;
396    walk(src, [&](const std::vector<size_t>& idx) {
397      const PointCloud<U>& cloud = src(idx);
398      if (cloud.is_indexed())
399        return;
400      const auto& coords = cloud.coords();
401      const void* source = static_cast<const void*>(coords.storage_data());
402      if (!castSources.contains(source))
403      {
404        PointCloud<T> casted(
405          std::vector<size_t>{coords.shape(0), coords.shape(1)});
406        for (size_t i = 0; i < coords.shape(0); ++i)
407        {
408          for (size_t j = 0; j < coords.shape(1); ++j)
409            casted(i, j) = static_cast<T>(coords(i, j));
410        }
411        castSources.emplace(source, std::move(casted));
412      }
