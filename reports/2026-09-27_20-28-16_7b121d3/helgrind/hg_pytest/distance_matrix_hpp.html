264      : m_storage(source.m_storage.view_with_layout(
265          DistanceMatrixLayout(source.m_storage.layout().source_size(),
266            std::move(indices))))
267    { }
268
269    void bounds_check(size_t i, size_t j) const
270    {
271      if (i >= size() || j >= size())
272      {
273        std::ostringstream oss;
274        oss << "DistanceMatrix index (" << i << ", " << j << ") out of range for matrix of size " << size();
275        throw std::out_of_range(oss.str());
276      }
277    }
278
279    FixedRankTensor<T, 2, DistanceMatrixLayout> m_storage;
280  };

283  struct is_compressed_matrix<DistanceMatrix<T>> : std::true_type {};
