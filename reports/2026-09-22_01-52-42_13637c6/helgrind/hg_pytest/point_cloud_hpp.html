20  /// A point cloud of shape (n_points, dim).
21  ///
22  /// A PointCloud either owns its coordinates or is an indexed view: it shares
23  /// another cloud's coordinate buffer and selects points through an attached
24  /// index set. Access via n_points()/dim()/operator()(i, j) is transparent to
25  /// which mode it is in, so consumers need no special case. This
26  /// lets a tensor of subsamples store one shared source plus small index arrays
27  /// instead of re-storing every (possibly high-dimensional) point.
28  ///
29  /// Deliberately not a Tensor<T>: the raw coordinate storage and the selected
30  /// points disagree for indexed views, so tensor-level access has no single
31  /// meaning here. Use the cloud-level members for the selected points, or
32  /// coords() to reach the underlying storage explicitly.

255    /// Materialize the selected points into a contiguous coordinate tensor.
256    /// Returns the coordinates as-is when not indexed.
257    [[nodiscard]] Tensor<T> materialize() const
258    {
259      if (!is_indexed())
260      {
261        return m_coords;
262      }
263
264      const size_t n = n_points();
265      const size_t d = dim();
266      Tensor<T> out({n, d});
267      for (size_t i = 0; i < n; ++i)
268      {
269        const auto row = static_cast<size_t>(m_indices(i));
270        for (size_t j = 0; j < d; ++j)
271        {

314  /**
315   * Cast a tensor of point clouds (Tensor<PointCloud<U>>) to a different precision
316   * (Tensor<PointCloud<T>>), converting each point cloud's coordinates.
317   *
318   * Dense coordinate sharing carries over: each distinct coordinate buffer is
319   * cast once. A transient indexed view is materialized because an ordinary
320   * tensor must contain self-contained point-cloud values.
321   */
322  template <typename T, typename U>
323  requires std::is_constructible_v<T, U>
324  [[nodiscard]] Tensor<PointCloud<T>> pcloud_cast(const Tensor<PointCloud<U>>& src)
325  {
326    Tensor<PointCloud<T>> result(src.shape());
327
328    // Cast each distinct source buffer once...
329    std::map<std::shared_ptr<const void>, Tensor<T>, std::owner_less<std::shared_ptr<const void>>> castSources;
330    walk(src, [&](const std::vector<size_t>& idx) {
331      const PointCloud<U>& cloud = src(idx);
332      const Tensor<U>& coords = cloud.coords();
333      if (coords.rank() == 0)
334      {
335        return;
336      }
337      if (cloud.is_indexed())
338      {
339        return;
340      }
341      if (!castSources.contains(coords.storage_owner()))
342      {
343        castSources.emplace(coords.storage_owner(), tensor_cast<T>(coords));
344      }
345    });
346
347    // ...then rebuild every cell on its shared cast source.
348    walk(src, [&](const std::vector<size_t>& idx) {
349      const PointCloud<U>& cloud = src(idx);
350      if (cloud.coords().rank() == 0)
351      {
352        result(idx) = PointCloud<T>();
353        return;
354      }
355      if (cloud.is_indexed())
356      {
357        result(idx) = PointCloud<T>(tensor_cast<T>(cloud.materialize()));
358      }
359      else
360      {
361        result(idx) = PointCloud<T>(
362          castSources.at(cloud.coords().storage_owner()));
363      }
364    });
365    return result;
366  }
