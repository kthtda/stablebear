143  /// Convert @p value from unit @p from to unit @p to. The factor is an
144  /// integer that multiplies or divides the value, so the result is rounded
145  /// once. The factor is exact unless it exceeds 2^53 (days and attoseconds).
146  /// A plain number, from Unitless, becomes a count of @p to unchanged; a
147  /// time in a unit cannot become a plain number.
148  [[nodiscard]] constexpr float64_t convert_time(float64_t value, TimeUnit from, TimeUnit to)
149  {
150    if (from == to || from == TimeUnit::Unitless)
151    {
152      return value;
153    }
154    if (to == TimeUnit::Unitless)
155    {
156      throw std::invalid_argument("Times in a unit cannot become plain numbers");
157    }
158    const auto& src = detail::time_unit_info(from);
159    const auto& dst = detail::time_unit_info(to);
160    const auto g = std::gcd(src.mult, dst.mult);
161    const int32_t diff = src.exp - dst.exp;
162    const float64_t num = static_cast<float64_t>(src.mult / g) * detail::exact_power_of_ten(std::max(diff, 0));
163    const float64_t den = static_cast<float64_t>(dst.mult / g) * detail::exact_power_of_ten(std::max(-diff, 0));
164    return den == 1 ? value * num : value / den;
165  }

255    /// The number of @p fine in one @p coarse, a unit at least as long, or
256    /// nullopt if it exceeds int64. It is a whole number, since the
257    /// multipliers of coarser units are multiples of finer ones.
258    constexpr std::optional<int64_t> units_per(const TimeUnitInfo& coarse, const TimeUnitInfo& fine)
259    {
260      int64_t ratio = coarse.mult / fine.mult;
261      for (int32_t k = fine.exp; k < coarse.exp; ++k)
262      {
263        if (ratio > std::numeric_limits<int64_t>::max() / 10)
264        {
265          return std::nullopt;
266        }
267        ratio *= 10;
268      }

273    /// 2^62 in the finer unit, so that a difference of two such fits in int64.

275    {
276      if (!factor)

280      const auto most = ((int64_t{1} << 62) - 1) / *factor;

285  /// The time from @p epoch to @p date in @p unit. When both counts fit
286  /// below 2^62 in the finest of the three units, the difference is taken
287  /// there in integers and converted to @p unit as convert_time() converts
288  /// it. Otherwise each count is split into whole days and a remainder in
289  /// its own unit, and the difference of the days and that of the
290  /// remainders, taken in integers when a day of the finer unit fits, are
291  /// converted to @p unit separately and added, so that no resolution or
292  /// span overflows. stablebear.timeseries computes
293  /// dates the same way, so that a series and a tensor of it agree.
294  /// Throws std::invalid_argument if a unit has no length, and
295  /// std::overflow_error if the days differ by more than int64 holds.
296  [[nodiscard]] inline float64_t time_since(const DateTime& date, const DateTime& epoch, TimeUnit unit)
297  {
298    if (date.unit == TimeUnit::Unitless || epoch.unit == TimeUnit::Unitless || unit == TimeUnit::Unitless)
299    {
300      throw std::invalid_argument("Dates, epochs, and the times between them need a unit with a length");
301    }
302    const auto& d = detail::time_unit_info(date.unit);
303    const auto& e = detail::time_unit_info(epoch.unit);
304    const auto& u = detail::time_unit_info(unit);
305    const auto& fine = detail::not_coarser(d, e) ? (detail::not_coarser(d, u) ? d : u)
306                                                 : (detail::not_coarser(e, u) ? e : u);
307    const auto dateFactor = detail::units_per(d, fine);
308    const auto epochFactor = detail::units_per(e, fine);
309    if (detail::fits_in_fine(date.count, dateFactor) && detail::fits_in_fine(epoch.count, epochFactor))
310    {
311      return convert_time(static_cast<float64_t>(date.count * *dateFactor - epoch.count * *epochFactor), fine.unit,
312                          unit);
313    }
314    // Each count as whole days and a remainder in its own unit.
315    const auto& days = detail::time_unit_info(TimeUnit::Days);
316    const auto split = [&](int64_t count, const detail::TimeUnitInfo& info) {
317      // A ratio beyond int64 exceeds every count, which is then all remainder.
318      const auto ratio = detail::units_per(days, info);
319      return ratio ? std::pair{count / *ratio, count % *ratio} : std::pair{int64_t{0}, count};
320    };
321    const auto [dateDays, dateRest] = split(date.count, d);
322    const auto [epochDays, epochRest] = split(epoch.count, e);
323    constexpr auto max = std::numeric_limits<int64_t>::max();
324    constexpr auto min = std::numeric_limits<int64_t>::min();
325    if ((epochDays > 0 && dateDays < min + epochDays) || (epochDays < 0 && dateDays > max + epochDays))
326    {
327      throw std::overflow_error("A date is too many days from the epoch to count");
328    }
329    const auto dayDifference = convert_time(static_cast<float64_t>(dateDays - epochDays), TimeUnit::Days, unit);
330    // The remainders, below a day each, subtract in int64 in the finer of
331    // their units when a day of it is below 2^62, as down to picoseconds.
332    const auto& finer = detail::not_coarser(d, e) ? d : e;
333    if (detail::fits_in_fine(1, detail::units_per(days, finer)))
334    {
335      const auto restDifference = dateRest * *detail::units_per(d, finer) - epochRest * *detail::units_per(e, finer);
336      return dayDifference + convert_time(static_cast<float64_t>(restDifference), finer.unit, unit);
337    }
338    return dayDifference + convert_time(static_cast<float64_t>(dateRest), date.unit, unit)
339           - convert_time(static_cast<float64_t>(epochRest), epoch.unit, unit);
340  }

677  template <typename T>
678  class TimeSeries
679  {
680  public:
681    using value_type = T;
682
683    /// An empty one-channel series in seconds, so that series can be tensor
684    /// elements.
685    TimeSeries()
686      : TimeSeries(Times(std::vector<float64_t>{}), Tensor<T>(std::vector<size_t>{0}), DefaultSnapTolerance)
687    {
688    }
689
690    /// A series of @p channels, whose times are in @p unit, counted from
691    /// @p epoch, or from none. A series without a unit has no epoch.
692    TimeSeries(std::vector<Channel<T>> channels, TimeUnit unit, std::optional<DateTime> epoch = std::nullopt);
693
694    /// A series with one channel.
695    TimeSeries(Times times, const Tensor<T>& values, float64_t snapTolerance, TimeUnit unit = TimeUnit::Seconds,
696               std::optional<DateTime> epoch = std::nullopt)
697      : TimeSeries({Channel<T>(std::move(times), values, snapTolerance)}, unit, epoch)
698    {
699    }
700
701    /// Combine one-channel series into one series; channel i is @p parts[i]
702    /// converted to @p unit, with its own snap settings. @p unit defaults to
703    /// the unit the parts share and is required when their units differ.
704    [[nodiscard]] static TimeSeries from_channels(const std::vector<TimeSeries>& parts,
705                                                  std::optional<TimeUnit> unit);
706
707    [[nodiscard]] const Channel<T>& channel(size_t i) const { return m_channels.at(i); }
708    [[nodiscard]] size_t n_channels() const noexcept { return m_channels.size(); }
709
710    /// The channels at @p indices, in that order, sharing their storage with
711    /// this series and queried in the same unit. Negative indices count from
712    /// the end.
713    [[nodiscard]] TimeSeries select(const std::vector<std::ptrdiff_t>& indices) const;
714    [[nodiscard]] TimeUnit unit() const noexcept { return m_unit; }
715
716    /// The instant time 0 stands for, if any.
717    [[nodiscard]] const std::optional<DateTime>& epoch() const noexcept { return m_epoch; }
718
719    /// The same series in @p unit, with every channel converted by
720    /// Channel::to(), sharing the observations. A series without a unit gets
721    /// @p unit: its numbers become counts of it.
722    [[nodiscard]] TimeSeries to(TimeUnit unit) const;
723
724    /// The same series counted from @p epoch, or from none, sharing the
725    /// observations; no time is shifted.
726    [[nodiscard]] TimeSeries with_epoch(std::optional<DateTime> epoch) const;
727
728    /// The same series with @p delta, in its unit, added to the times of
729    /// every channel, counted from @p epoch, or from none.
730    [[nodiscard]] TimeSeries shifted(float64_t delta, std::optional<DateTime> epoch) const;
731
732    /// The times at which every channel can be evaluated: from the latest
733    /// first timestamp to the earliest end, a channel's last timestamp or its
734    /// end, as check_observed() compares them.
735    using Domain = TimeDomain;
736
737    /// The domain of the series. Requires every channel to have
738    /// observations.
739    [[nodiscard]] Domain domain() const;
740
741    /// The same unit, epoch, and channels.
742    [[nodiscard]] bool operator==(const TimeSeries& other) const = default;
743
744    /// The same unit, and channels with the same observations; see
745    /// Channel::same_observations(). The epoch is not compared.
746    [[nodiscard]] bool same_observations(const TimeSeries& other) const
747    {
748      return m_unit == other.m_unit && std::ranges::equal(m_channels, other.m_channels, [](const auto& a, const auto& b) {
749               return a.same_observations(b);
750             });
751    }
752
753    /// Every channel's value at @p t, given in the series' unit: one leaf per
754    /// channel, in channel order, each a tensor of that channel's shape.
755    /// @p t must lie between the first and last observations of every
756    /// channel, both included.
757    [[nodiscard]] NestedTensor<T> evaluate(float64_t t) const;
758
759    /// Every channel's values at the times @p t, given in the series' unit:
760    /// one leaf per channel, in channel order, each a tensor of shape
761    /// (*t.shape(), *shape). The queries are sorted once, in O(m log m) for m
762    /// queries, and merged against each channel's timestamps, galloping
763    /// between queries: at most O(m log n) and O(m + n) for n observations.
764    /// Every time must lie in the domain.
765    [[nodiscard]] NestedTensor<T> evaluate(const Tensor<float64_t>& t) const;
766
767  private:
768    /// Throw TimeSeriesDomainError, naming the domain, unless @p t, in the
769    /// series' unit, lies in the domain. The check is exact: it compares
770    /// against the timestamps converted to the series' unit, not the query
771    /// converted to each channel's unit.
772    void check_observed(float64_t t) const;
773
774    /// Throw unless a series in @p unit can have @p epoch.
775    static void check_epoch(TimeUnit unit, const std::optional<DateTime>& epoch)
776    {
777      if (epoch && unit == TimeUnit::Unitless)
778      {
779        throw std::invalid_argument("A series without a unit cannot have an epoch; give its times as durations or dates");
780      }
781    }

783    std::vector<Channel<T>> m_channels;
784    TimeUnit m_unit;
785    std::optional<DateTime> m_epoch;
786  };
